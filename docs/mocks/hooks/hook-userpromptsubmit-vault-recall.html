#!/usr/bin/env bun
/**
 * UserPromptSubmit, synchronous: Jev picks the wiki notes that bear on the
 * prompt and they are injected as additionalContext. Silent when the prompt is
 * short, nothing matches, no note reaches 0.85, Jev fails, or `aleph judge haiku`.
 */
import { readJsonOr } from "../lib/json.ts";
import { mkdirSync, readdirSync, rmSync, statSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { judgeChoice, nouls, type Noul } from "../lib/jev.ts";
import { truncate } from "../lib/otlp.ts";
import { postTurnSpan } from "./lib/trace.ts";
import { claimLine, recallWiki } from "../vault/lib/index.ts";
import { noteRead, recordRun } from "../vault/lib/vault.ts";
import { scopeOfCwd, vaultDir } from "../lib/paths.ts";
import { errText } from "../lib/errtext.ts";
import { isAgent } from "../lib/env.ts";

const MIN_PROMPT = 20;
// Jev took about 1,670 ms for 40 questions and 1,250 ms for 20 (4 Nov), against a 1,500 ms timeout,
// so the candidates go in parallel requests of at most PER_REQUEST questions.
const PER_REQUEST = 20;
const MAX_ASKED = 100;
const MAX_INJECTED = 3;
const CUT = 0.85;
const JEV_TIMEOUT_MS = 1500;
const SEEN_DAYS = 7;

/** The notes this session's hook already injected, in `<vault>/.index/seen/<session>.json`. */
const seenFile = (root: string, session: string) => join(root, ".index", "seen", `${session.replace(/[^\w-]/g, "_")}.json`);

function readSeen(root: string, session: string | undefined): Set<string> {
  if (!session) return new Set();
  return new Set(readJsonOr<string[]>(seenFile(root, session), []));
}

/** Add `rels` to the session's list, and delete the lists older than SEEN_DAYS. */
function writeSeen(root: string, session: string | undefined, seen: Set<string>, rels: string[]) {
  const dir = join(root, ".index", "seen");
  mkdirSync(dir, { recursive: true });
  try {
    for (const f of readdirSync(dir)) if (statSync(join(dir, f)).mtimeMs < Date.now() - SEEN_DAYS * 86400_000) rmSync(join(dir, f), { force: true });
  } catch (e) { recordRun(root, "seen prune", "failed", errText(e)); throw e; }
  recordRun(root, "seen prune", "ok");
  if (session) writeFileSync(seenFile(root, session), JSON.stringify([...seen, ...rels]));
}

/** One `vault-recall` span on the turn's trace. A failed post changes nothing. */
async function postSpan(input: any, start: number, meta: { candidates: number; asked: number; kept: number; capped: boolean; error?: string }, offered: string[], context: string, askedPaths: string[] = []) {
  await postTurnSpan(input, "vault-recall", start, {
    "langfuse.observation.type": "span",
    "langfuse.observation.output": context ? truncate(context) : undefined,
    "langfuse.observation.metadata.candidates": meta.candidates,
    "langfuse.observation.metadata.asked": meta.asked,
    "langfuse.observation.metadata.kept": meta.kept,
    "langfuse.observation.metadata.capped": meta.capped,
    "langfuse.observation.metadata.error": meta.error,
    "langfuse.observation.metadata.askedPaths": askedPaths.length ? askedPaths.join("\n") : undefined,
    "langfuse.observation.metadata.offered": offered.length ? offered.join("\n") : undefined,
  });
}

async function main() {
  if (isAgent()) return;
  const input = JSON.parse((await Bun.stdin.text()) || "{}");
  const prompt: string = typeof input.prompt === "string" ? input.prompt.trim() : "";
  if (prompt.length < MIN_PROMPT || judgeChoice() === "haiku") return;
  const start = Date.now();
  const root = vaultDir();
  const seen = readSeen(root, input.session_id);
  const { ranked, core } = await recallWiki(root, prompt, scopeOfCwd(input.cwd ?? process.cwd()));
  // A core note's line is already in the start context.
  const skip = new Set([...seen, ...core.map((n) => n.rel)]);
  const all = ranked.map((r) => r.note).filter((n) => !skip.has(n.rel));
  const candidates = all.slice(0, MAX_ASKED);
  const meta: Parameters<typeof postSpan>[2] = { candidates: all.length, asked: candidates.length, kept: 0, capped: all.length > MAX_ASKED };
  if (!candidates.length) { await postSpan(input, start, meta, [], ""); return; }
  const criteria = { true: "The note's subject is something the prompt asks about or depends on.", false: "The note is about something else." };
  const batches = Array.from({ length: Math.ceil(candidates.length / PER_REQUEST) }, (_, b) => candidates.slice(b * PER_REQUEST, (b + 1) * PER_REQUEST));
  const settled = await Promise.allSettled(batches.map((batch) => {
    const questions: Record<string, Noul> = {};
    batch.forEach((n, i) => {
      questions[`n${i}`] = { type: "noul", instructions: `Does note \`${n.title}\` (its claim line: ${claimLine(n)}) bear on \`prompt\`?`, criteria };
    });
    return nouls({ prompt }, questions, JEV_TIMEOUT_MS);
  }));
  const errors: string[] = [];
  const scored: { n: (typeof all)[number]; p: number }[] = [];
  settled.forEach((s, b) => {
    if (s.status === "rejected") { errors.push(errText(s.reason)); return; }
    batches[b].forEach((n, i) => scored.push({ n, p: s.value.p[`n${i}`] }));
  });
  if (errors.length) meta.error = errors.join("; ");
  const picked = scored.filter((x) => x.p >= CUT).sort((a, b) => b.p - a.p).slice(0, MAX_INJECTED);
  meta.kept = picked.length;
  const lines = picked.map(({ n }) => `- ${n.title} (${join(root, n.rel)}): ${claimLine(n)}`);
  const additionalContext = picked.length ? ["Vault notes that may bear on this prompt (Jev):", ...lines].join("\n") : "";
  await postSpan(input, start, meta, picked.map(({ n }) => n.rel), additionalContext, candidates.map((n) => n.rel));
  if (!picked.length) return;
  writeSeen(root, input.session_id, seen, picked.map(({ n }) => n.rel));
  noteRead(root, picked.map(({ n }) => n.title), prompt, false);
  console.log(JSON.stringify({ hookSpecificOutput: { hookEventName: "UserPromptSubmit", additionalContext } }));
}

// The factory's JSON state files. The only module that parses one: a missing file is the caller's fallback, a file
// that does not parse is a bug and throws. The write goes through a rename, so a reader never sees half a file.
import { mkdirSync, readFileSync, renameSync, writeFileSync } from "node:fs";
import { dirname } from "node:path";

/** The JSON in `file`, or `fallback` when there is no file. A file that does not parse throws. */
export function readJsonOr<T>(file: string, fallback: T): T {
  let text: string;
  try { text = readFileSync(file, "utf8"); } catch (e) {
    if ((e as NodeJS.ErrnoException).code === "ENOENT") return fallback;
    throw e;
  }
  return JSON.parse(text);
}

/** The temp file a writer fills beside `file` before it renames it over `file`: named by pid, so two processes do not collide. */
export const tmpName = (file: string): string => `${file}.${process.pid}.tmp`;

/** Writes `value` to `file` as JSON: to a temp file, then a rename over `file`. `mode` sets the new file's permissions. */
export function writeJson(file: string, value: unknown, mode?: number): void {
  mkdirSync(dirname(file), { recursive: true });
  const tmp = tmpName(file);
  writeFileSync(tmp, JSON.stringify(value, null, 2) + "\n", { mode });
  renameSync(tmp, file);
}

/**
 * Jev, TypeSafe's decision model, through OpenRouter's decisions endpoint.
 * It answers typed questions (choice, noul, score) with probabilities and no
 * text, at $0.042 per million input tokens (OpenRouter, 27 Oct 2026). It
 * cannot be called through chat/completions.

import { conn, getJson, putJson } from "./db.ts";
import { openRouterKey } from "./env.ts";
import { logSetting } from "./layout.ts";

const JEV_MODEL = "typesafe/jev-1.13";
type JudgeChoice = "jev" | "haiku";

/**
 * Whether Jev screens before Haiku, everywhere aleph uses it: ALEPH_JUDGE,
 * else the `judge` setting that `aleph judge` writes, else jev.
 */
export function judgeChoice(): JudgeChoice {
  const v = process.env.ALEPH_JUDGE ?? getJson("settings", "judge", "");
  return v === "haiku" ? "haiku" : "jev";
}

export function setJudgeChoice(choice: JudgeChoice) {
  putJson("settings", "judge", choice);
  logSetting(`judge is ${choice}`);
}
const URL = () => process.env.ALEPH_JEV_URL ?? "https://openrouter.ai/api/alpha/decisions";

export type Noul = { type: "noul"; instructions: string; criteria?: { true: string; false: string } };
export type Question = { type: "choice"; instructions: string; criteria: Record<string, string> } | Noul;
type Answer =
  | { type: "choice"; choice: string; probabilities?: Record<string, number>; confidence?: number }
  | { type: "noul"; noul: number };
export interface Decision { answers: Record<string, Answer>; cost: number; model: string; ms: number }

/** One request: the state and the questions in, typed answers out. Throws on any failure. */
export async function decide(state: unknown, questions: Record<string, Question>, timeoutMs = 15_000): Promise<Decision> {
  const key = openRouterKey();
  if (!key) throw new Error("no OPENROUTER_API_KEY");
  const started = performance.now();
  const res = await fetch(URL(), {
    method: "POST",
    headers: { authorization: `Bearer ${key}`, "content-type": "application/json" },
    body: JSON.stringify({ model: JEV_MODEL, state, questions }),
    signal: AbortSignal.timeout(timeoutMs),
  });
  const body: any = await res.json().catch(() => null);
  if (!res.ok) throw new Error(`jev ${res.status}: ${JSON.stringify(body?.error ?? body).slice(0, 200)}`);
  for (const id of Object.keys(questions)) {
    const a = body?.answers?.[id];
    if (a?.type !== questions[id].type) throw new Error(`jev: no ${questions[id].type} answer for ${id}`);
  }
  const cost = body.usage?.cost ?? 0;
  // One row per call in `jev`: OpenRouter bills Jev on aleph's key, and its calls are in no transcript, so the OpenRouter rate (models.ts) subtracts them.
  if (cost > 0) {
    try { conn().query("INSERT INTO jev (time, cost) VALUES (?, ?)").run(new Date().toISOString(), cost); } catch { /* a lost row costs cents */ }
  }
  return { answers: body.answers, cost, model: body.model ?? JEV_MODEL, ms: performance.now() - started };
}

/** decide() for yes-or-no questions: each question's probability of yes, by id. */
export async function nouls(state: unknown, questions: Record<string, Noul>, timeoutMs?: number): Promise<{ p: Record<string, number>; cost: number }> {
  const d = await decide(state, questions, timeoutMs);
  const p = Object.fromEntries(Object.keys(questions).map((id) => [id, (d.answers[id] as { noul: number }).noul]));
  return { p, cost: d.cost };
}

export function truncate(value: unknown, max = MAX_FIELD): string {
  const text = typeof value === "string" ? value : JSON.stringify(value) ?? "";
  if (text.length <= max) return text;
  return `${text.slice(0, max)}…[+${text.length - max} chars]`;
}

function encode(value: AttrValue): Record<string, unknown> {
  if (Array.isArray(value)) return { arrayValue: { values: value.map((item) => ({ stringValue: String(item) })) } };
  if (typeof value === "boolean") return { boolValue: value };
  if (typeof value === "number") return Number.isInteger(value) ? { intValue: String(value) } : { doubleValue: value };
  return { stringValue: value };
}

export function attrs(record: Record<string, AttrValue | undefined | null>): OtlpAttr[] {
  return Object.entries(record)
    .filter(([, value]) => value !== undefined && value !== null && value !== "")
    .map(([key, value]) => ({ key, value: encode(value as AttrValue) }));
}

/**
 * The derived search index and the one ranker that `vault recall` and the
 * UserPromptSubmit hook share.
 *
 * The index is an FTS5 table at `<vault>/.index/vault.sqlite`, kept out of git.
 * It is rebuilt when the vault HEAD differs from the HEAD stored in it. A vault
 * that is not a git repo has no HEAD, so its index is rebuilt on each open.
 */
import { Database } from "bun:sqlite";
import { existsSync, mkdirSync, renameSync, rmSync } from "node:fs";
import { join } from "node:path";
import { coreNotes, loadWiki, scopeOf, screenWords as words, type Note } from "./vault.ts";
import { probe } from "../../lib/git.ts";
import { tmpName } from "../../lib/json.ts";

/** The reciprocal rank fusion constant. */
const K = 60;
/** The bm25 hits one search reads. */
const FTS_LIMIT = 200;

/** The first paragraph of a note's body on one line. */
export const claimLine = (n: Pick<Note, "body">): string => n.body.trim().split(/\n\s*\n/)[0].replace(/\s+/g, " ").trim();

/** A note as the index returns it: no path and no text. */
type Hit = Pick<Note, "rel" | "title" | "fm" | "body" | "wiki" | "archived">;

const aliasesOf = (n: Pick<Note, "fm">): string[] => {
  const a = n.fm.aliases;
  return (Array.isArray(a) ? a : a ? [a] : []).map((x) => String(x).toLowerCase());
};

function head(root: string): string {
  const p = probe(root, "rev-parse", "HEAD");
  return p.ok ? p.out : "";
}

// The suffix names the columns of the table: an index at the same HEAD with fewer is rebuilt.
const stamp = (at: string): string => (at ? `${at}:fm` : "");

/** The open index when it holds the vault HEAD `at`, else null. */
function current(root: string, at: string): Database | null {
  const file = join(root, ".index", "vault.sqlite");
  if (!at || !existsSync(file)) return null;
  const db = new Database(file);
  try {
    if ((db.query("select value from meta where key = 'head'").get() as { value: string } | null)?.value === at) return db;
  } catch (e) {
    // A vault.sqlite with no meta table is rebuilt; a busy or corrupt file is not.
    if (!/no such table/.test((e as Error).message)) { db.close(); throw e; }
  }
  db.close();
  return null;
}

/** The index for `notes` at the vault HEAD `at`, built beside the file and renamed in. */
function build(root: string, at: string, notes: Note[]): Database {
  // A hook that reads at the same time sees the old index or the new one.
  const dir = join(root, ".index");
  const file = join(dir, "vault.sqlite");
  mkdirSync(dir, { recursive: true });
  const tmp = tmpName(file);
  rmSync(tmp, { force: true });
  const db = new Database(tmp);
  db.run("create table meta (key text primary key, value text)");
  db.run(`create virtual table notes using fts5(title, aliases, claim, body, rel unindexed, scope unindexed, kind unindexed, updated unindexed, fm unindexed, tokenize = "porter unicode61")`);
  const add = db.prepare("insert into notes values (?, ?, ?, ?, ?, ?, ?, ?, ?)");
  db.transaction(() => {
    for (const n of notes) add.run(n.title, aliasesOf(n).join(" "), claimLine(n), n.body, n.rel, scopeOf(n), String(n.fm.kind ?? ""), String(n.fm.updated ?? ""), JSON.stringify(n.fm));
    db.run("insert into meta values ('head', ?)", [at]);
  })();
  db.close();
  renameSync(tmp, file);
  return new Database(file);
}

/** The index for `notes`, rebuilt first when the vault HEAD moved. */
function openIndex(root: string, notes: Note[]): Database {
  const at = stamp(head(root));
  return current(root, at) ?? build(root, at, notes);
}

/** The rel paths of the notes that match any query word, best bm25 first. A query with no word of four letters, such as `bun`, searches on all its tokens. */
function search(db: Database, query: string): string[] {
  const long = words(query);
  const ws = [...new Set(long.length ? long : query.toLowerCase().match(/[a-z0-9]+/g) ?? [])];
  if (!ws.length) return [];
  const match = ws.map((w) => `"${w}"`).join(" OR ");
  return (db.query("select rel from notes where notes match ? order by bm25(notes, 10, 5, 3, 1) limit ?").all(match, FTS_LIMIT) as { rel: string }[]).map((r) => r.rel);
}

/**
 * How a note's title and aliases match the query: 0 the title is the query,
 * 1 an alias is, 2 the title or an alias holds the query or one of its words.
 * Undefined when none of these.
 */
function titleTier(n: Hit, q: string, want: Set<string>): { tier: number; hits: number } | undefined {
  const title = n.title.toLowerCase(), aliases = aliasesOf(n);
  if (title === q) return { tier: 0, hits: Infinity };
  if (aliases.includes(q)) return { tier: 1, hits: Infinity };
  if (title.includes(q) || aliases.some((a) => a.includes(q))) return { tier: 2, hits: Infinity };
  const have = new Set(words([n.title, ...aliases].join(" ")));
  const hits = [...want].filter((w) => have.has(w)).length;
  return hits ? { tier: 2, hits } : undefined;
}

interface Ranked<T = Note> { note: T; rank: number }

/**
 * The index for `notes`, searched for `query` among `among` (all of `notes` by default) and closed again.
 * The ranking is `rankNotes`'s.
 */
export function recall(root: string, notes: Note[], query: string, scope?: string, among: Note[] = notes): Ranked[] {
  return recallAll(root, notes, [query], scope, among)[0];
}

/** `recall` for each of `queries`, in order, on one open of the index. */
export function recallAll(root: string, notes: Note[], queries: string[], scope?: string, among: Note[] = notes): Ranked[][] {
  const db = openIndex(root, notes);
  try { return queries.map((q) => rankNotes(db, among, q, scope)); } finally { db.close(); }
}

/**
 * The notes that match `query`, fused by reciprocal rank (k = 60) over three
 * lists: the title and alias matches, the bm25 hits, and, with `scope`, that
 * scope's notes by `updated`, newest first. Only a note on one of the first
 * two lists is returned; the third list orders them and adds none.
 * `rank` is the title tier (0, 1 or 2), or 3 for a body-only hit.
 */
function rankNotes<T extends Hit>(db: Database, notes: T[], query: string, scope?: string): Ranked<T>[] {
  const q = query.trim().toLowerCase();
  if (!q) return [];
  const want = new Set(words(q));
  const byRel = new Map(notes.map((n) => [n.rel, n]));
  const tiers = new Map<string, number>();
  const titled = notes.flatMap((n) => { const t = titleTier(n, q, want); return t ? [{ n, ...t }] : []; })
    .sort((a, b) => a.tier - b.tier || b.hits - a.hits || a.n.title.localeCompare(b.n.title));
  for (const { n, tier } of titled) tiers.set(n.rel, tier);
  const lists: string[][] = [titled.map(({ n }) => n.rel), search(db, q).filter((rel) => byRel.has(rel))];
  if (scope !== undefined) {

function walk(dir: string, out: string[] = []): string[] {
  if (!existsSync(dir)) return out;
  for (const e of readdirSync(dir, { withFileTypes: true })) {
    if (e.name.startsWith(".")) continue;
    const p = join(dir, e.name);
    // A symlink is followed, as statSync did.
    const isDir = e.isSymbolicLink() ? statSync(p).isDirectory() : e.isDirectory();
    if (isDir) walk(p, out);
    else if (e.name.endsWith(".md")) out.push(p);
  }
  return out;
}

/** Parsed notes by root and path, valid while a file's mtime and size are unchanged. */
const cache = new Map<string, { mtimeMs: number; size: number; note: Note }>();

/** The notes under `dir`, parsed in parallel; a file whose mtime and size match the cache is not parsed again. */
async function load(root: string, dir: string): Promise<Note[]> {
  const seen = new Set<string>();
  const notes = await Promise.all(
    walk(dir).map(async (p) => {
      const key = `${root}\0${p}`;
      seen.add(key);
      const st = await stat(p);
      const hit = cache.get(key);
      if (hit && hit.mtimeMs === st.mtimeMs && hit.size === st.size) return { ...hit.note };
      const note = parseNote(p, root, await readFile(p, "utf8"));
      cache.set(key, { mtimeMs: st.mtimeMs, size: st.size, note });
      return { ...note };
    }),
  );
  for (const key of cache.keys()) if (key.startsWith(`${root}\0${dir}`) && !seen.has(key)) cache.delete(key);
  return notes;
}

/**
 * How stale a note's claim is, or null when it is inside its window.
 *
 * The window is the kind's decay budget, widened or narrowed by confidence,
 * and it runs from `updated` only. A recall proves a note is used, not that it
 * is true, so a read does not reset the window; the maintainer's re-check does.
 */
function staleness(note: Note, now = new Date()): string | null {
  const kind = String(note.fm.kind ?? "");
  if (!(KINDS as readonly string[]).includes(kind)) return null;
  const updated = String(note.fm.updated ?? "");
  if (!/^\d{4}-\d{2}-\d{2}$/.test(updated)) return null;
  const window = Math.round(DECAY_DAYS[kind as Kind] * (CONFIDENCE_FACTOR[String(note.fm.confidence)] ?? 1));
  const days = Math.floor((now.getTime() - Date.parse(updated)) / 86400_000);
  if (days <= window) return null;
  return `${kind}/${note.fm.confidence} wants a re-check every ${window} days; updated ${updated}, ${days} days ago`;

    const wb = words([b.title, ...listOf(b, "aliases")].join(" "));
    const shared = [...wa].filter((w) => wb.has(w));
    if (shared.length >= 2) pairs.push({ note: a.title, rule: "overlap", other: b.title, detail: `shares "${shared.join('", "')}" with ${b.title} in scope ${a.fm.scope}; neither supersedes the other`, score: shared.length / new Set([...wa, ...wb]).size });
  }
  // Most similar first (Jaccard of the word sets), ties by title, so a tied pair keeps one place and one
  // direction however the filesystem lists the notes.
  warn.push(...pairs.sort((x, y) => y.score - x.score || x.note.localeCompare(y.note) || x.other!.localeCompare(y.other!)).map(({ score, ...f }) => f));
  return { refuse, warn };
}

/**
 * The repairs that need no judgement. `--fix` edits frontmatter and never the
 * body: frontmatter is structural and each defect below has one right answer,
 * where a missing `## Evidence` or `as of` marker is a claim about the world.
 * Appending those headings would turn "this claim is unbacked" into a passing
 * check and lose the only way to find an unbacked claim later.
 *
 * Edits are line-level, so a note keeps its own formatting and the diff stays
 * readable. Returns null when there is nothing to repair.
 */
export function fixFrontmatter(note: Note, added: string | null): { text: string; repairs: string[] } | null {
  if (!note.hasFrontmatter) return null;
  const { frontmatter, body } = splitFrontmatter(note.text);
  const lines = frontmatter!.split(/\r?\n/);
  const repairs: string[] = [];

  for (const key of LISTS) {
    const v = note.fm[key];
    if (v === undefined || Array.isArray(v)) continue;
    const i = lines.findIndex((l) => new RegExp(`^${key}:\\s`).test(l));
    // Rebuild the line from its own text, not from the parsed value, and only
    // when the text is a plain scalar. A line the parser read as a string
    // because it could not read it at all is a defect to report, not to wrap.
    const raw = i < 0 ? "" : lines[i].slice(key.length + 1).trim();
    if (i < 0 || !raw || /["'\[\]#]/.test(raw)) continue;
    lines[i] = `${key}: [${raw}]`;
    repairs.push(`${key} wrapped in a list`);
  }
  if (note.fm.supersedes === undefined) { lines.push("supersedes: []"); repairs.push("added supersedes: []"); }
  if (note.fm.updated === undefined && added) { lines.push(`updated: ${added}`); repairs.push(`added updated: ${added}, the date the note entered git`); }

  if (!repairs.length) return null;
  return { text: `---\n${lines.join("\n")}\n---\n${body}`, repairs };
}

/** Body text that names a date the reader cannot resolve. Reported, never rewritten. */
export function relativeDates(notes: Note[]): Finding[] {
  const words = /\b(yesterday|today|tomorrow|last (?:night|week|month|year)|this (?:morning|week|month)|next week|a few days ago|recently|just now)\b/gi;
  const out: Finding[] = [];
  for (const n of wikiNotes(notes)) {
    // Code names a word, it does not date anything: `today()` is a function.
    const prose = n.body.replace(/```[\s\S]*?```/g, "").replace(/`[^`\n]+`/g, "");
    const hits = [...new Set([...prose.matchAll(words)].map((m) => m[0].toLowerCase()))];
    if (hits.length) out.push({ note: n.title, rule: "relative-date", detail: `says ${hits.map((h) => `"${h}"`).join(", ")}; give the date` });
  }
  return out;
}

/**
 * Rewrite the `scope:` line of every note in one scope.
 *
 * Line-level, like `fixFrontmatter`, so a note keeps its own formatting. The
 * vault carried five scope names for two projects, because a rename moved the
 * repo and left every note behind: `voice-bridge` and `voice-bridge-mcp` are
 * both Sidetone, and `caller` is Voiceover. Nothing catches that, because
 * `scope` is free text.
 *
 * Returns the notes it would change, each with its new text. It touches no
 * other line and invents no scope.
 */
export function renameScope(notes: Note[], from: string, to: string): { note: Note; text: string }[] {
  const out: { note: Note; text: string }[] = [];
  for (const n of wikiNotes(notes)) {
    if (String(n.fm.scope) !== from) continue;
    const { frontmatter, body } = splitFrontmatter(n.text);
    if (frontmatter === null) continue;
    const lines = frontmatter.split(/\r?\n/);
    const i = lines.findIndex((l) => /^scope:\s/.test(l));
    if (i < 0) continue;
    lines[i] = `scope: ${to}`;
    out.push({ note: n, text: `---\n${lines.join("\n")}\n---\n${body}` });
  }
  return out.sort((a, b) => a.note.rel.localeCompare(b.note.rel));
}

/**
 * The SessionStart context for `scope`: MEMORY.md whole, then the core lines,
 * cut at whole lines to START_BUDGET bytes. `cut` counts the core lines left
 * out; the last line then points to the rest.
 */
export function startContext(root: string, notes: Note[], scope: string): { text: string; cut: number } {
  const memory = notes.find((n) => n.rel === "MEMORY.md")?.text?.trim();
  const lines = coreNotes(notes, scope).map((n) => `- ${String(n.fm.core).trim()} [[${n.title}]]`);
  const render = (kept: string[]) => {
    const cut = lines.length - kept.length;
    return [`<vault path="${root}">`, ...(memory ? [memory, ""] : []), ...kept, ...(cut ? [`${cut} more core lines: aleph memory recall --scope ${scope}`] : []), "</vault>"].join("\n");
  };
  let kept = lines;
  while (kept.length && Buffer.byteLength(render(kept)) > START_BUDGET) kept = kept.slice(0, -1);
  return { text: render(kept), cut: lines.length - kept.length };
}

 * Only LANGFUSE_* keys, plus any `extra` names the caller asks for (the UI server asks for its password): the file also holds model API keys, and a nested `claude -p`

 */
export function loadDotenv(path = envFile()): void {
  let text: string;
  try { text = readFileSync(path, "utf8"); } catch { return; }
  for (const raw of text.split("\n")) {
    const line = raw.trim();
    if (!line || line.startsWith("#")) continue;
    const eq = line.indexOf("=");

/**

 * say `test`), else `headless` when an ancestor is `claude -p`, else `interactive`.
 * The parent chain is the only place a hook can see how claude was started.
 */
export function classifyEnvironment(ancestorArgvs: string[][], override = process.env.ALEPH_ENV): string {
  if (override) return override;
  // claude is argv[0], or argv[1] behind a shell wrapper; only flags after it count
  const headless = ancestorArgvs.some((argv) => {
    const at = argv.slice(0, 2).findIndex((arg) => arg === "claude" || arg.endsWith("/claude"));
    return at >= 0 && argv.slice(at + 1).some((arg) => arg === "-p" || arg === "--print");
  });

}

function ancestorArgvs(): string[][] {

  for (let i = 0; i < DEPTH && pid > 1; i++) {

}
