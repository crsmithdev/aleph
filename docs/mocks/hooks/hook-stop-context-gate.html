#!/usr/bin/env bun
/**
 * The context gate. docs/context-gate.md is the contract.
 *
 *   Stop  fill below the warn fraction     → pass silently
 *         above it                         → pass, warn Chris
 *         session has compacted            → block once, ask for a handoff
 *   A headless session (a factory worker, `claude -p`) has no Chris to warn or hand off to: pass silently.
 */
import { once } from "./lib/handshake.ts";
import { assess, read, threshold } from "../lib/session-context.ts";
import { jobId } from "../lib/env.ts";

const input = JSON.parse(await Bun.stdin.text());
if (input.hook_event_name !== "Stop") process.exit(0);
if (jobId() || process.env.CLAUDE_CODE_ENTRYPOINT?.startsWith("sdk")) process.exit(0);

const sessionId: string | undefined = input.session_id;
const transcript: string = input.transcript_path ?? "";
if (!sessionId || !transcript) process.exit(0);

const gated = once(`ctx:${sessionId}`);
const verdict = assess(read(transcript), threshold(), gated.seen(), input.stop_hook_active === true);

if (verdict.kind === "warn") console.log(JSON.stringify({ systemMessage: verdict.message }));
if (verdict.kind === "gate") {
  gated.mark();
  console.log(JSON.stringify({ decision: "block", reason: `Context gate: ${verdict.reason}` }));
}

/**
 * Hooks are separate processes. A Pre* event leaves a small file that the
 * matching Post* event picks up, so a span can carry its real start time and
 * its parent. Files live in ~/.aleph/spool (ALEPH_SPOOL overrides, for tests).
 * Each kind of record has its own typed slot below; the key prefix is the kind.
 */
import { readJsonOr } from "../../lib/json.ts";
import { mkdirSync, readdirSync, statSync, unlinkSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { alephPath } from "../../lib/paths.ts";
import type { Snapshot } from "./snapshot.ts";

export function spoolDir(): string {
  return alephPath("ALEPH_SPOOL", "spool");
}

function pathFor(key: string): string {
  return join(spoolDir(), `${key.replace(/[^A-Za-z0-9_.:-]/g, "_")}.json`);
}

function slot<T>(key: string) {
  const get = (): T | null => {
    return readJsonOr<T | null>(pathFor(key), null);
  };
  return {
    get,
    set(value: T): void {
      mkdirSync(spoolDir(), { recursive: true });
      writeFileSync(pathFor(key), JSON.stringify(value));
    },
    take(): T | null {
      const value = get();
      if (value !== null) { try { unlinkSync(pathFor(key)); } catch { /* already gone */ } }
      return value;
    },
  };
}

/** Left by SessionStart: the name, tags and cwd every turn trace repeats, and the last turn's trace for a rating. */
export interface Session { start: number; name?: string; tags?: string[]; cwd?: string; lastTraceId?: string }
export const session = (id: string) => slot<Session>(`session:${id}`);

/** The turn's start and prompt, from UserPromptSubmit to Stop. */
export const turn = (promptId: string) => slot<{ start: number; input: string }>(`turn:${promptId}`);

/** A tool call's start and input, from PreToolUse to PostToolUse. */
export const tool = (toolUseId: string) => slot<{ start: number; input: unknown }>(`tool:${toolUseId}`);

/** A subagent's span and trace, from SubagentStart to SubagentStop. */
export const agent = (agentId: string) => slot<{ start: number; spanId: string; parentSpanId?: string; traceId: string }>(`agent:${agentId}`);

/** The repo's snapshot at the prompt, for the verify gate at Stop. */
export const snap = (promptId: string) => slot<Snapshot>(`snap:${promptId}`);

/** A count of denials: the verify gate's of a turn, or the refit-code hook's blocks of a session. */
export function denials(promptId: string) {
  const s = slot<number>(`denials:${promptId}`);
  return {
    count: (): number => s.get() ?? 0,
    add: (): void => s.set((s.get() ?? 0) + 1),
    clear: (): void => void s.take(),
  };
}

/** A mark for something that happens once per key, such as a block once per session. */
export function once(key: string) {
  const s = slot<number>(key);
  return { seen: (): boolean => s.get() !== null, mark: (): void => s.set(Date.now()) };
}

const MAX_AGE_MS = 86_400_000;

/** A Pre event whose Post never came leaves a file behind; drop anything older than a day. */
export function prune(): void {
  let names: string[];
  try { names = readdirSync(spoolDir()); } catch { return; }
  const cutoff = Date.now() - MAX_AGE_MS;
  for (const name of names) {
    const path = join(spoolDir(), name);
    try { if (statSync(path).mtimeMs < cutoff) unlinkSync(path); } catch { /* raced with another hook */ }
  }
}

/**
 * How full the session's context is, read from the transcript, and what to do
 * about it. The window comes from the same places Claude Code reads it:
 * CLAUDE_CODE_AUTO_COMPACT_WINDOW, then the autoCompactWindow setting, then the
 * model default of 1M. Compaction fires at window - min(maxOutput, 20k) - 13k
 * (measured in 2.1.283); 33k is that headroom.
 *
 * Only compact_boundary counts as compaction. A microcompact_boundary evicts
 * old tool results and leaves the conversation intact, so the context is still
 * the conversation rather than a summary of one; it also fires early in a
 * tool-heavy session and lowers the fill, which would gate a healthy session at
 * a low reading.
 *
 * The budget is what the setting asks for, not a prediction. Measured over 99
 * sessions on 2026-09-27: three compacted within 12k of the threshold and five
 * ran past it untouched, one to 314k. The warn names the budget for that
 * reason.
 */
import { readJsonOr } from "./json.ts";
import { homedir } from "node:os";
import { join } from "node:path";
import { entriesBackward, readText } from "./jsonl.ts";

const HEADROOM = 33_000;
const DEFAULT_WINDOW = 1_000_000;
export const DEFAULT_WARN = 0.75;

export interface Reading {
  fill: number;
  compacted: boolean;
}

/** The last assistant request's context size, and whether the session has compacted. Both are read from the end, parsing only the lines that can answer. */
export function read(transcriptPath: string): Reading {
  const text = readText(transcriptPath);
  let fill = 0;
  for (const e of entriesBackward(text, /"usage"/)) {
    const usage = e.type === "assistant" && !e.isSidechain ? e.message?.usage : undefined;
    if (!usage) continue;
    fill = (usage.input_tokens ?? 0) + (usage.cache_read_input_tokens ?? 0) + (usage.cache_creation_input_tokens ?? 0);
    break;
  }
  let compacted = false;
  for (const e of entriesBackward(text, /"compact_boundary"|"isCompactSummary"/)) {
    if (e.isSidechain || !(e.isCompactSummary || (e.type === "system" && e.subtype === "compact_boundary"))) continue;
    compacted = true;
    break;
  }
  return { fill, compacted };
}

/** Claude Code's settings file: ALEPH_CLAUDE_SETTINGS, else ~/.claude/settings.json. */
export const claudeSettings = (): string => process.env.ALEPH_CLAUDE_SETTINGS ?? join(homedir(), ".claude", "settings.json");

/** The context window Claude Code runs with. */
export function window(settingsPath = claudeSettings()): number {
  const env = Number(process.env.CLAUDE_CODE_AUTO_COMPACT_WINDOW);
  if (Number.isFinite(env) && env > 0) return env;
  const configured = Number(readJsonOr<any>(settingsPath, {}).autoCompactWindow);
  return Number.isFinite(configured) && configured > 0 ? configured : DEFAULT_WINDOW;
}

/** Tokens the session can hold before Claude Code compacts it. */
export function threshold(settingsPath = claudeSettings()): number {
  return Math.max(1000, window(settingsPath) - HEADROOM);
}

export type Verdict =
  | { kind: "quiet" }
  | { kind: "warn"; message: string }
  | { kind: "gate"; reason: string };

/**
 * Quiet below the warn fraction. A warning to Chris above it. A block once the
 * session has compacted: from there on the context is a summary of a summary,
 * and a handoff written now is better than one written later.
 */
export function assess(reading: Reading, limit: number, blocked: boolean, retry: boolean): Verdict {
  const k = (n: number) => `${Math.round(n / 1000)}k`;
  if (reading.compacted && !blocked && !retry) {
    return {
      kind: "gate",
      reason: `this session has compacted, so its context is now a summary. Write the handoff with the aleph:handoff skill, then tell Chris to run /clear; the next session picks the handoff up. Do not start new work in this session.`,
    };
  }
  const fraction = Number(process.env.ALEPH_CONTEXT_WARN ?? DEFAULT_WARN);
  const warnAt = limit * (Number.isFinite(fraction) && fraction > 0 ? fraction : DEFAULT_WARN);
  if (reading.fill >= warnAt && !reading.compacted) {
    return { kind: "warn", message: `Context ${k(reading.fill)} of the ${k(limit)} budget. /aleph:handoff then /clear keeps the next turn sharp.` };
  }
  return { kind: "quiet" };
}

import { readFileSync } from "node:fs";
import { alephPath } from "./paths.ts";

export interface LangfuseConfig {
  baseUrl: string;
  publicKey: string;
  secretKey: string;
}

/** True in a factory agent: the agent settings in jobs/lib/routing/models.ts set ALEPH_AGENT for the claude process of each role. */
export const isAgent = (): boolean => Boolean(process.env.ALEPH_AGENT);

/** The id of the job this process runs for, if any: the unit sets ALEPH_JOB_ID (jobs/lib/roles/unit.ts), and its children inherit it. */
export const jobId = (): string | undefined => process.env.ALEPH_JOB_ID;

/** The dotenv file: ALEPH_ENV_FILE, else ~/.aleph/.env. */
const envFile = (): string => alephPath("ALEPH_ENV_FILE", ".env");

/** One key from the dotenv file, never copied to process.env; null when the file or the key is missing. */
function dotenvKey(name: string): string | null {
  let text: string;
  try { text = readFileSync(envFile(), "utf8"); } catch { return null; }
  const m = text.match(new RegExp(`^\\s*(?:export\\s+)?${name}=(.*)$`, "m"));
  return m ? m[1].trim().replace(/^(["'])(.*)\1$/, "$2") : null;
}

/** OPENROUTER_API_KEY from the environment, else that one key from the dotenv file; null when neither has it. */
export function openRouterKey(): string | null {
  return process.env.OPENROUTER_API_KEY || dotenvKey("OPENROUTER_API_KEY");
}

/** ANTHROPIC_API_KEY from the dotenv file only: a process that has it in its environment bills the API instead of the subscription. */
export const anthropicKey = (): string | null => dotenvKey("ANTHROPIC_API_KEY");

/**
 * Hooks inherit the shell, not a dotenv; read the dotenv file (`envFile()`) ourselves. process.env wins.
 * Only LANGFUSE_* keys, plus any `extra` names the caller asks for (the UI server asks for its password): the file also holds model API keys, and a nested `claude -p`
 * that inherits ANTHROPIC_API_KEY bills the API instead of the subscription.
 */
export function loadDotenv(path = envFile()): void {
  let text: string;
  try { text = readFileSync(path, "utf8"); } catch { return; }
  for (const raw of text.split("\n")) {
    const line = raw.trim();
    if (!line || line.startsWith("#")) continue;
    const eq = line.indexOf("=");
    if (eq < 1) continue;
    const key = line.slice(0, eq).trim();
    if (!key.startsWith("LANGFUSE_")) continue;
    let value = line.slice(eq + 1).trim();
    if ((value.startsWith('"') && value.endsWith('"')) || (value.startsWith("'") && value.endsWith("'"))) value = value.slice(1, -1);
    if (process.env[key] === undefined) process.env[key] = value;
  }
}

export function langfuseConfig(): LangfuseConfig | null {
  loadDotenv();
  const publicKey = process.env.LANGFUSE_PUBLIC_KEY;
  const secretKey = process.env.LANGFUSE_SECRET_KEY;
  if (!publicKey || !secretKey) return null;
  return { baseUrl: (process.env.LANGFUSE_BASE_URL ?? "http://127.0.0.1:3010").replace(/\/+$/, ""), publicKey, secretKey };
}

/**
 * Langfuse environment for the session: `ALEPH_ENV` if set (the live tests
 * say `test`), else `headless` when an ancestor is `claude -p`, else `interactive`.
 * The parent chain is the only place a hook can see how claude was started.
 */
export function classifyEnvironment(ancestorArgvs: string[][], override = process.env.ALEPH_ENV): string {
  if (override) return override;
  // claude is argv[0], or argv[1] behind a shell wrapper; only flags after it count
  const headless = ancestorArgvs.some((argv) => {
    const at = argv.slice(0, 2).findIndex((arg) => arg === "claude" || arg.endsWith("/claude"));
    return at >= 0 && argv.slice(at + 1).some((arg) => arg === "-p" || arg === "--print");
  });
  return headless ? "headless" : "interactive";
}

const DEPTH = 6;

function ancestorArgvs(): string[][] {
  const out: string[][] = [];
  let pid = process.ppid;
  for (let i = 0; i < DEPTH && pid > 1; i++) {
    let cmdline: string, status: string;
    try { cmdline = readFileSync(`/proc/${pid}/cmdline`, "utf8"); status = readFileSync(`/proc/${pid}/status`, "utf8"); } catch { break; }
    out.push(cmdline.split("\0").filter(Boolean));
    pid = Number(status.match(/^PPid:\s*(\d+)/m)?.[1] ?? 0);
  }
  return out;
}

export function sessionEnvironment(): string {
  return classifyEnvironment(ancestorArgvs());
}
