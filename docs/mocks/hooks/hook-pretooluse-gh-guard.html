#!/usr/bin/env bun
/**
 * PreToolUse on Bash: a `gh` command that writes to a GitHub repo the user
 * does not own asks first. A post there reaches other people. Reads go
 * through. The target is the repo named by `-R`/`--repo`, by a github.com URL,
 * by `repos/<owner>/<repo>` in `gh api`, or else the origin of the directory
 * the command runs in. A write whose target cannot be found asks. The user is
 * `ALEPH_GH_USER`, or else the login of `gh api user`, asked once a day.
 */
/*
 * A hook that exits 1 does not block the tool, so a guard that throws before
 * its verdict exits 2 (deny) with its name and the error on stderr.
 */
import { join, resolve } from "node:path";
import { GH_USER_CMD } from "../lib/env.ts";
import { today } from "../lib/day.ts";
import { probe } from "../lib/git.ts";
import { readJsonOr, writeJson } from "../lib/json.ts";
import { spoolDir } from "./lib/handshake.ts";

try {
  const input = JSON.parse(await Bun.stdin.text());
  const command: string = input.tool_input?.command ?? "";
  if (!/\bgh\s/.test(command)) process.exit(0);

  const VERBS = new Set([
    "create", "comment", "edit", "close", "reopen", "delete", "merge", "review", "fork", "lock", "unlock",
    "pin", "unpin", "transfer", "ready", "rename", "archive", "unarchive", "sync", "set", "develop",
    "rerun", "cancel", "upload", "add", "remove", "enable", "disable", "run",
  ]);
  const WORD = `"[^"]*"|'[^']*'|[^\\s;&|()]+`;
  const unquote = (w: string) => w.replace(/^(["'])(.*)\1$/, "$2");

  /** The owner of a github.com remote URL, or null. */
  function remoteOwner(dir: string | null): string | null {
    if (!dir) return null;
    return probe(dir, "remote", "get-url", "origin").out.match(/github\.com[:/]([^/\s]+)\//)?.[1] ?? null;
  }

  /** The owner a write in `part` goes to, "" for a read, or null when the target is unknown. */
  function target(part: string, dir: string | null): string | null | "" {
    const m = part.match(/\bgh\s+(\S+)(?:\s+(\S+))?/);
    if (!m) return "";
    const [, group, verb] = m;
    if (group === "api") {
      const method = part.match(/(?:-X|--method)[\s=]+["']?(\w+)/)?.[1]?.toUpperCase();
      const fields = /\s(?:-[fF]|--field|--raw-field|--input)[\s=]/.test(part);
      const write = method ? method !== "GET" : fields && !/\bgraphql\b/.test(part);
      if (!write && !/\bmutation\b/.test(part)) return "";
      return part.match(/\brepos\/([^/\s"']+)\//)?.[1] ?? null;
    }
    if (!verb || !VERBS.has(verb)) return "";
    const named = part.match(new RegExp(`(?:-R|--repo)[\\s=]+(${WORD})`))?.[1];
    if (named) return unquote(named).replace(/^(?:https?:\/\/)?github\.com\//, "").split("/")[0];
    const url = part.match(/github\.com\/([^/\s"']+)\//)?.[1];
    if (url) return url;
    const positional = group === "repo" ? part.match(/\bgh\s+repo\s+\S+\s+([\w.-]+)\/[\w.-]+/)?.[1] : undefined;
    return positional ?? remoteOwner(dir);
  }

  let dir: string | null = input.cwd ?? process.cwd();
  const owners: (string | null)[] = [];
  for (const part of command.split(/&&|\|\||[;|\n]/)) {
    const cd = part.match(new RegExp(`^\\s*cd\\s+(${WORD})`))?.[1];
    if (cd !== undefined) {
      const path = unquote(cd);
      dir = dir === null || (cd[0] !== "'" && /[$`]/.test(path)) ? null : resolve(dir, path.replace(/^~(?=\/|$)/, process.env.HOME ?? "~"));
      continue;
    }
    const owner = target(part, dir);
    if (owner !== "") owners.push(owner);
  }
  if (owners.length === 0) process.exit(0);

  /** The login of `gh api user`: a network call, so a login is kept in the spool for the day. */
  function ghLogin(): string {
    const file = join(spoolDir(), "gh-user.json");
    const day = today();
    const kept = readJsonOr<{ day: string; login: string } | null>(file, null);
    if (kept?.day === day) return kept.login;
    const login = Bun.spawnSync(GH_USER_CMD, { stdout: "pipe", stderr: "ignore" }).stdout.toString().trim();
    if (login) writeJson(file, { day, login });
    return login;
  }

  const user = process.env.ALEPH_GH_USER || ghLogin();
  const foreign = owners.filter((o) => o === null || o.toLowerCase() !== user.toLowerCase());
  if (foreign.length === 0) process.exit(0);

  const named = foreign.map((o) => o ?? "an unknown owner").join(", ");
  console.log(JSON.stringify({ hookSpecificOutput: {
    hookEventName: "PreToolUse",
    permissionDecision: "ask",
    permissionDecisionReason: `gh-guard: this command writes to a GitHub repo of ${named}, not of ${user || "the user"}. A post there reaches other people.`,
  } }));
} catch (e) {
  console.error(`gh-guard: the guard failed (${e instanceof Error ? (e.stack ?? e.message) : String(e)}), so this tool call is denied. Fix the hook and retry.`);
  process.exit(2);
}

/**
 * Hooks inherit the shell, not a dotenv; read the dotenv file (`envFile()`) ourselves. process.env wins.
 * Only LANGFUSE_* keys, plus any `extra` names the caller asks for (the UI server asks for its password): the file also holds model API keys, and a nested `claude -p`
 * that inherits ANTHROPIC_API_KEY bills the API instead of the subscription.
 */
export function loadDotenv(path = envFile()): void {
  let text: string;
  try { text = readFileSync(path, "utf8"); } catch { return; }
  for (const raw of text.split("\n")) {
    const line = raw.trim();
    if (!line || line.startsWith("#")) continue;
    const eq = line.indexOf("=");
    if (eq < 1) continue;
    const key = line.slice(0, eq).trim();
    if (!key.startsWith("LANGFUSE_")) continue;
    let value = line.slice(eq + 1).trim();
    if ((value.startsWith('"') && value.endsWith('"')) || (value.startsWith("'") && value.endsWith("'"))) value = value.slice(1, -1);
    if (process.env[key] === undefined) process.env[key] = value;
  }
}

export function langfuseConfig(): LangfuseConfig | null {
  loadDotenv();
  const publicKey = process.env.LANGFUSE_PUBLIC_KEY;
  const secretKey = process.env.LANGFUSE_SECRET_KEY;
  if (!publicKey || !secretKey) return null;
  return { baseUrl: (process.env.LANGFUSE_BASE_URL ?? "http://127.0.0.1:3010").replace(/\/+$/, ""), publicKey, secretKey };
}

/**
 * Langfuse environment for the session: `ALEPH_ENV` if set (the live tests
 * say `test`), else `headless` when an ancestor is `claude -p`, else `interactive`.
 * The parent chain is the only place a hook can see how claude was started.
 */
export function classifyEnvironment(ancestorArgvs: string[][], override = process.env.ALEPH_ENV): string {
  if (override) return override;
  // claude is argv[0], or argv[1] behind a shell wrapper; only flags after it count
  const headless = ancestorArgvs.some((argv) => {
    const at = argv.slice(0, 2).findIndex((arg) => arg === "claude" || arg.endsWith("/claude"));
    return at >= 0 && argv.slice(at + 1).some((arg) => arg === "-p" || arg === "--print");
  });
  return headless ? "headless" : "interactive";
}

const DEPTH = 6;

function ancestorArgvs(): string[][] {
  const out: string[][] = [];
  let pid = process.ppid;
  for (let i = 0; i < DEPTH && pid > 1; i++) {
    let cmdline: string, status: string;
    try { cmdline = readFileSync(`/proc/${pid}/cmdline`, "utf8"); status = readFileSync(`/proc/${pid}/status`, "utf8"); } catch { break; }
    out.push(cmdline.split("\0").filter(Boolean));
    pid = Number(status.match(/^PPid:\s*(\d+)/m)?.[1] ?? 0);
  }
  return out;
}

export function sessionEnvironment(): string {
  return classifyEnvironment(ancestorArgvs());
}

/**
 * git for aleph: the one runner (a spawn that says where it failed, aleph's env, and a wait on another
 * process's index.lock), a fetch, a tree with files replaced, and the main checkout following origin.
 * Every write to main goes through a push that git refuses when main moved, so the push is the compare-and-swap.
 */
import { existsSync, rmSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";

// A worker's env carries GIT_CONFIG_* to send its pushes nowhere. aleph's own
// writes (an issue commit from a worker) are not the worker's push.
const baseEnv = (): Record<string, string | undefined> => {
  const env: Record<string, string | undefined> = { ...process.env };
  for (const k of Object.keys(env)) if (k.startsWith("GIT_CONFIG_")) delete env[k];
  return env;
};

/** `Bun.spawnSync` for git. A spawn that throws is rethrown with the original as `cause` and a one-line message that says where and with which PATH it ran. */
export function spawnGit(cmd: string[], opts: { cwd?: string; env?: Record<string, string | undefined>; stdin?: "ignore" | Blob; stdout: "pipe"; stderr: "pipe" | "ignore" }): { exitCode: number | null; stdout: Buffer; stderr: Buffer } {
  try {
    return Bun.spawnSync(cmd, opts) as { exitCode: number | null; stdout: Buffer; stderr: Buffer };
  } catch (e) {
    const cwd = opts.cwd;
    let here: string;
    try { here = process.cwd(); } catch (err) { here = `unreadable (${(err as Error).message})`; }
    const PATH = (opts.env ?? process.env).PATH;
    const found = Bun.which("git", { PATH: PATH ?? "" });
    const text = (e as Error).message.replace(/\s+/g, " ");
    throw new Error(
      `git could not start (${text}). The cwd passed is ${cwd === undefined ? "none" : JSON.stringify(cwd)}` +
      `${cwd === undefined ? "" : ` and ${existsSync(cwd) ? "exists" : "does not exist"} (existsSync ${existsSync(cwd)})`}. ` +
      `process.cwd() is ${here}. The PATH passed is ${JSON.stringify(PATH)}. Bun.which finds git at ${JSON.stringify(found)}. ` +
      `TMPDIR is ${JSON.stringify(process.env.TMPDIR)} and tmpdir() ${existsSync(tmpdir()) ? "exists" : "does not exist"} (existsSync ${existsSync(tmpdir())}).`,
      { cause: e },
    );
  }
}

export function git(cwd: string, ...args: string[]): { ok: boolean; out: string } {
  return gitWith({}, cwd, ...args);
}

function gitWith(o: { env?: Record<string, string>; stdin?: string | Uint8Array }, cwd: string, ...args: string[]): { ok: boolean; out: string } {
  // Several facts gates write the vault in the same second; a git call that
  // meets another's index.lock waits and tries again (6 facts lost to it, 1 to 3 Nov).
  for (let i = 0; ; i++) {
    const p = spawnGit(["git", ...args], {
      cwd, env: { ...baseEnv(), ...o.env }, stdin: o.stdin === undefined ? "ignore" : new Blob([o.stdin as BlobPart]), stdout: "pipe", stderr: "pipe",
    });
    const r = { ok: p.exitCode === 0, out: (p.stdout.toString() + (p.exitCode === 0 ? "" : p.stderr.toString())).trimEnd() };
    if (r.ok || !r.out.includes("index.lock") || i === 10) return r;
    Bun.sleepSync(200 * (i + 1));
  }
}

/** `git` in `cwd`, not ok (and no throw) when `cwd` does not exist. */
export function probe(cwd: string, ...args: string[]): { ok: boolean; out: string } {
  return existsSync(cwd) ? git(cwd, ...args) : { ok: false, out: "" };
}

/** The ref of a repo's main to read: `origin/<main>` when the repo has it, else the local `<main>`. */
export function mainRef(cwd: string, main: string): string {
  return git(cwd, "rev-parse", "--verify", "-q", `origin/${main}^{commit}`).ok ? `origin/${main}` : main;
}

/** The pause between two tries of a step that met a held git lock: `sleep` waits, Bun.sleepSync unless the caller passes Bun.sleep and awaits it. A test replaces it to free the lock. */
export const lockRetry = { pause: (sleep: (ms: number) => unknown = Bun.sleepSync): unknown => sleep(100 + Math.random() * 200) };

type Run = { ok: boolean; out: string };

/** The tries of a fetch: it yields the number of each try and takes its result. A try after the first waits out the ref lock that another process holds (up to 5 tries). */
function* fetchTries(): Generator<number, Run, Run> {
  let r = yield 0;
  for (let i = 1; i < 5 && !r.ok && r.out.includes("cannot lock ref"); i++) r = yield i;
  return r;
}

/** `git fetch -q origin <refs>`, tried again (up to 5 times) when another process holds the ref lock. Returns the last result. */
export function fetchOrigin(cwd: string, ...refs: string[]): Run {
  const tries = fetchTries();
  for (let t = tries.next(); ; ) {
    if (t.done) return t.value;
    if (t.value) lockRetry.pause();
    t = tries.next(git(cwd, "fetch", "-q", "origin", ...refs));
  }
}

/** fetchOrigin that does not block: each try is an async spawn and each pause an async sleep. */
export async function fetchOriginAsync(cwd: string, ...refs: string[]): Promise<Run> {
  const tries = fetchTries();
  for (let t = tries.next(); ; ) {
    if (t.done) return t.value;
    if (t.value) await lockRetry.pause(Bun.sleep);
    let p;
    // A spawn throws when cwd is gone: the UI's timer then fetches a store a test removed.
    try { p = Bun.spawn(["git", "fetch", "-q", "origin", ...refs], { cwd, env: baseEnv(), stdin: "ignore", stdout: "ignore", stderr: "pipe" }); }
    catch (e) { return { ok: false, out: (e as Error).message }; }
    const [err, code] = await Promise.all([new Response(p.stderr).text(), p.exited]);
    t = tries.next({ ok: code === 0, out: code === 0 ? "" : err.trimEnd() || `git fetch exited ${code}` });
  }
}

/** `tree` with each path in `files` set to its content, or removed when it is null. */
export function treeWithFiles(cwd: string, tree: string, files: Record<string, string | Uint8Array | null>): string {
  const index = join(tmpdir(), `aleph-index-${process.pid}-${crypto.randomUUID()}`);
  const env = { GIT_INDEX_FILE: index };
  try {
    const steps = [["read-tree", tree]];
    for (const [path, content] of Object.entries(files)) {
      if (content === null) { steps.push(["update-index", "--force-remove", path]); continue; }
      const blob = gitWith({ stdin: content }, cwd, "hash-object", "-w", "--stdin");
      if (!blob.ok) throw new Error(`hash-object: ${blob.out}`);
      steps.push(["update-index", "--add", "--cacheinfo", `100644,${blob.out},${path}`]);
    }
    for (const args of steps) {
      const r = gitWith({ env }, cwd, ...args);
      if (!r.ok) throw new Error(`${args[0]}: ${r.out}`);
    }
    const w = gitWith({ env }, cwd, "write-tree");
    if (!w.ok) throw new Error(`write-tree: ${w.out}`);
    return w.out;
  } finally {
    rmSync(index, { force: true });
  }
}

/**
 * Fast-forward the main checkout to origin/<main>. Returns why it could not,
 * or null. It never moves a checkout that is on another branch or has tracked changes.
 */
export function followMain(path: string, main: string): string | null {
  const head = git(path, "symbolic-ref", "--short", "HEAD").out;
  if (head !== main) return `main checkout not updated: it is on ${head || "a detached HEAD"}`;
  const dirty = git(path, "status", "--porcelain", "--untracked-files=no").out.split("\n").filter(Boolean).map((l) => l.slice(3));
  if (dirty.length) return `main checkout not updated: it has changes in ${dirty.join(", ")}`;
  const ff = git(path, "merge", "--ff-only", "-q", `origin/${main}`);
  return ff.ok ? null : `main checkout not updated: ${ff.out.split("\n")[0]}`;
}

// The factory's JSON state files. The only module that parses one: a missing file is the caller's fallback, a file
// that does not parse is a bug and throws. The write goes through a rename, so a reader never sees half a file.
import { mkdirSync, readFileSync, renameSync, writeFileSync } from "node:fs";
import { dirname } from "node:path";

/** The JSON in `file`, or `fallback` when there is no file. A file that does not parse throws. */
export function readJsonOr<T>(file: string, fallback: T): T {
  let text: string;
  try { text = readFileSync(file, "utf8"); } catch (e) {
    if ((e as NodeJS.ErrnoException).code === "ENOENT") return fallback;
    throw e;
  }
  return JSON.parse(text);
}

/** The temp file a writer fills beside `file` before it renames it over `file`: named by pid, so two processes do not collide. */
export const tmpName = (file: string): string => `${file}.${process.pid}.tmp`;

/** Writes `value` to `file` as JSON: to a temp file, then a rename over `file`. `mode` sets the new file's permissions. */
export function writeJson(file: string, value: unknown, mode?: number): void {
  mkdirSync(dirname(file), { recursive: true });
  const tmp = tmpName(file);
  writeFileSync(tmp, JSON.stringify(value, null, 2) + "\n", { mode });
  renameSync(tmp, file);
}

/**
 * Hooks are separate processes. A Pre* event leaves a small file that the
 * matching Post* event picks up, so a span can carry its real start time and
 * its parent. Files live in ~/.aleph/spool (ALEPH_SPOOL overrides, for tests).
 * Each kind of record has its own typed slot below; the key prefix is the kind.
 */
import { readJsonOr } from "../../lib/json.ts";
import { mkdirSync, readdirSync, statSync, unlinkSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { alephPath } from "../../lib/paths.ts";
import type { Snapshot } from "./snapshot.ts";

export function spoolDir(): string {
  return alephPath("ALEPH_SPOOL", "spool");
}

function pathFor(key: string): string {
  return join(spoolDir(), `${key.replace(/[^A-Za-z0-9_.:-]/g, "_")}.json`);
}

function slot<T>(key: string) {
  const get = (): T | null => {
    return readJsonOr<T | null>(pathFor(key), null);
  };
  return {
    get,
    set(value: T): void {
      mkdirSync(spoolDir(), { recursive: true });
      writeFileSync(pathFor(key), JSON.stringify(value));
    },
    take(): T | null {
      const value = get();
      if (value !== null) { try { unlinkSync(pathFor(key)); } catch { /* already gone */ } }
      return value;
    },
  };
}

const MAX_AGE_MS = 86_400_000;

/** A Pre event whose Post never came leaves a file behind; drop anything older than a day. */
export function prune(): void {
  let names: string[];
  try { names = readdirSync(spoolDir()); } catch { return; }
  const cutoff = Date.now() - MAX_AGE_MS;
  for (const name of names) {
    const path = join(spoolDir(), name);
    try { if (statSync(path).mtimeMs < cutoff) unlinkSync(path); } catch { /* raced with another hook */ }
  }
}
