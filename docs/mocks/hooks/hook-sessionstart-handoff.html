#!/usr/bin/env bun
/**
 * SessionStart on startup and /clear: show the user the summary line, else the
 * Intent line, of each handoff that waits for this session, and tell Claude to pick it up on the
 * first prompt. A handoff waits for the session when it is the project's
 * handoff at <handoffs>/projects/<scope>.md, or a named handoff at
 * <handoffs>/<name>.md whose `cwd:` line is the session's directory or one
 * that holds it. The full handoff texts and the session directory's
 * `git status` and `git log -5` go in the context when they stay within
 * 10,000 characters. Inside Herdr (HERDR_ENV=1, HERDR_PANE_ID set) the project
 * handoff of this pane is <handoffs>/projects/<scope>/<pane>.md, and the
 * hook lists each other file of that folder whose pane Herdr no longer has
 * as orphaned: Claude asks before it picks one up. There the flat file is
 * picked up by its path, as pickup with no argument reads the pane's file.
 * Inside the Herdr desktop app (HC_TERMINAL_ID set, and no Herdr
 * pane) the same holds for <scope>/hc-<HC_TERMINAL_ID>.md: the hook lists each
 * other hc- file of that folder whose terminal `hc get` does not find as orphaned.
 * Each runtime ignores the other's files.
 * No handoff, no output.
 * Headless (`claude -p`) sessions and factory jobs are skipped.
 */
import { readFileSync, statSync } from "node:fs";
import { resolve } from "node:path";
import { handoffs, intent, type Handoff } from "../lib/handoffs.ts";
import { scopeOfCwd } from "../lib/paths.ts";
import { probe } from "../lib/git.ts";
import { jobId } from "../lib/env.ts";
import { clock } from "../lib/day.ts";

const input = JSON.parse((await Bun.stdin.text()) || "{}");
if (jobId() || process.env.CLAUDE_CODE_ENTRYPOINT?.startsWith("sdk")) process.exit(0);

const cwd = resolve(input.cwd ?? process.cwd());
const scope = scopeOfCwd(cwd);
const all = handoffs();
const own = all.find((h) => h.kind === "project" && h.scope === scope);
const under = (text: string) => {
  const at = text.match(/^cwd:\s*(.+)$/m)?.[1].trim();
  return !!at && (cwd === resolve(at) || cwd.startsWith(`${resolve(at)}/`));
};
const pane = process.env.HERDR_ENV === "1" ? process.env.HERDR_PANE_ID : undefined;
const live = pane ? Bun.spawnSync(["herdr", "pane", "list"], { stdout: "pipe", stderr: "ignore" }) : undefined;
const has = (id: string) => new RegExp(`(^|[^\\w:-])${id.replace(/\W/g, "\\$&")}([^\\w:-]|$)`).test(live!.stdout.toString());
const term = !pane ? process.env.HC_TERMINAL_ID : undefined;
const me = pane ?? (term ? `hc-${term}` : undefined);
// A pane's file is <pane>.md, a desktop terminal's is hc-<terminal>.md; each runtime sees only its own.
const panes = me ? all.filter((h) => h.kind === "pane" && h.scope === scope && !!pane !== h.name.startsWith("hc-")) : [];
const gone = (id: string) => pane ? !has(id) : Bun.spawnSync(["hc", "get", id.slice(3)], { stdout: "ignore", stderr: "ignore" }).exitCode !== 0;
const orphans = pane && live?.exitCode !== 0 ? [] : panes.filter((h) => h.name !== me && gone(h.name));
const pick = (h: Handoff, orphan = false) => ({ path: h.path, name: h.kind === "named" ? h.name : "", orphan, text: readFileSync(h.path, "utf8") });
const waiting = [
  ...panes.filter((h) => h.name === me).map((h) => pick(h)),
  ...(own ? [pick(own)] : []),
  ...all.filter((h) => h.kind === "named").map((h) => pick(h)).filter((h) => under(h.text)),
  ...orphans.map((h) => pick(h, true)),
];
if (!waiting.length) process.exit(0);

const when = (path: string) => {
  const at = statSync(path).mtime;
  return at.toDateString() === new Date().toDateString() ? clock(at) : `${at.toLocaleDateString("en-US", { month: "short", day: "numeric" })} ${clock(at)}`;
};

Do not run the aleph:pickup skill on your own; if the user says yes, run it with the argument ${h.path}.` : `A handoff ${h.name ? `named ${h.name}` : "for this project"} waits at ${h.path}: ${intent(h.text)}. On the user's first prompt, run the aleph:pickup skill ${h.name || (me && h.path === own?.path) ? `with the argument ${h.name || h.path}` : "with no argument"} before anything else, unless that prompt is plainly about other work; then do that work and say in one line that the handoff still waits.`).join("\n");
const status = probe(cwd, "status");
const gitPart = status.ok ? `\n\ngit status:\n${status.out}\n\ngit log -5:\n${probe(cwd, "log", "-5").out}` : "";
const texts = waiting.map((h) => `${h.path}:\n\n${h.text.trim()}`).join("\n\n");
const full = `${short}\n\nThe handoff text follows. The pickup skill needs no Read of the file.\n\n${texts}${gitPart}`;
const additionalContext = full.length > 10000 ? short : full;

console.log(JSON.stringify({
  systemMessage: waiting.map((h) => `handoff ${h.orphan ? "orphaned" : "waiting"} (${h.name || scope}, ${when(h.path)}): ${intent(h.text)}`).join("\n"),
  hookSpecificOutput: {
    hookEventName: "SessionStart",
    additionalContext,
  },
}));

/**
 * The handoffs folder's layout, read in one place. A key is a path in the folder: `projects/<scope>.md` is a
 * project's handoff, `projects/<scope>/<id>.md` a Herdr pane's (`<pane>`) or a desktop terminal's (`hc-<terminal>`),
 * `<YYYY-MM-DD-HHMMSS>[-<name>].md` a handoff pickup archived, and any other `<name>.md` a named handoff.
 */
import { existsSync, readdirSync, statSync } from "node:fs";
import { join, relative, resolve } from "node:path";
import { clip } from "./naming.ts";
import { handoffsDir } from "./paths.ts";

type Kind = "project" | "pane" | "named" | "dated";
/** One handoff. `scope` is a project's or a pane's repo, `name` a pane's id or a named handoff's name, `day` an archived one's date; else "". */
export type Handoff = { key: string; path: string; kind: Kind; scope: string; name: string; day: string };

const PROJECT = /^projects\/([^/.][^/]*)\.md$/;
const PANE = /^projects\/([^/.][^/]*)\/([^/.][^/]*)\.md$/;
const DATED = /^(\d{4}-\d\d-\d\d)-\d{6}(?:-[^/]*)?\.md$/;
const NAMED = /^([^/.][^/]*)\.md$/;

/** The handoff at `at`, a key or a path in the folder; null for any other file. */
export function handoffOf(at: string): Handoff | null {
  const dir = resolve(handoffsDir()), key = relative(dir, resolve(dir, at)), path = join(dir, key);
  const project = PROJECT.exec(key), pane = PANE.exec(key), dated = DATED.exec(key), named = NAMED.exec(key);
  if (project) return { key, path, kind: "project", scope: project[1], name: "", day: "" };
  if (pane) return { key, path, kind: "pane", scope: pane[1], name: pane[2], day: "" };
  if (dated) return { key, path, kind: "dated", scope: "", name: "", day: dated[1] };
  return named && { key, path, kind: "named", scope: "", name: named[1], day: "" };
}

/** Every handoff in the folder, by key. */
export function handoffs(): Handoff[] {
  const dir = handoffsDir();
  const ls = (rel: string): string[] => (existsSync(join(dir, rel)) ? readdirSync(join(dir, rel)).map((f) => (rel ? `${rel}/${f}` : f)) : []);
  const scopes = ls("projects").filter((k) => statSync(join(dir, k)).isDirectory());
  return [...ls(""), ...ls("projects"), ...scopes.flatMap(ls)].sort().flatMap((k) => handoffOf(k) ?? []);
}

/** A handoff's `summary:` line, else the first non-blank line under its `## Intent`, else its first line. */
export function intent(text: string): string {
  const summary = text.match(/^summary:[ \t]*(.+)/m)?.[1].trim();
  if (summary) return clip(summary, 160);
  const lines = text.split("\n");
  const at = lines.findIndex((l) => /^##\s+intent\s*$/i.test(l));
  const from = at < 0 ? lines : lines.slice(at + 1);
  return clip(from.find((l) => l.trim() && !/^#/.test(l)) ?? "", 160);
}

/**
 * The user's live store, from the passwd entry of this uid, so a test that sets HOME cannot move it. Bun 1.4.2's
 * `os.userInfo().homedir` follows HOME (Node's does not), so the file is read. Null when no entry is found.
 */
function passwdHome(): string | null {
  const uid = String(process.getuid?.());
  try {
    const row = readFileSync("/etc/passwd", "utf8").split("\n").map((l) => l.split(":")).find((f) => f[2] === uid);
    return row?.[5] || null;
  } catch { return null; }
}

/**

export function alephPath(name: string, rel: string): string {

}

/** The registry key of the repo that `cwd` is in, a worktree of it too; else the folder name of the main checkout. With `git` false: the key whose path holds `cwd`, else "", and no git run. */
export function scopeOfCwd(cwd: string, git = true): string {
  touch("registry");
  const repos = readProjects().filter((p) => p.fields.path).map((p) => ({ key: p.name, path: expand(p.fields.path) }));
  const at = (dir: string) => repos.find((r) => dir === r.path || dir.startsWith(`${r.path.replace(/\/+$/, "")}/`));
  if (!git) return at(cwd)?.key ?? "";
  const common = probe(cwd, "rev-parse", "--path-format=absolute", "--git-common-dir");
  const root = common.ok ? dirname(common.out) : cwd;
  return (at(cwd) ?? at(root))?.key ?? basename(root);
}

/**
 * git for aleph: the one runner (a spawn that says where it failed, aleph's env, and a wait on another
 * process's index.lock), a fetch, a tree with files replaced, and the main checkout following origin.
 * Every write to main goes through a push that git refuses when main moved, so the push is the compare-and-swap.
 */
import { existsSync, rmSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";

// A worker's env carries GIT_CONFIG_* to send its pushes nowhere. aleph's own
// writes (an issue commit from a worker) are not the worker's push.
const baseEnv = (): Record<string, string | undefined> => {
  const env: Record<string, string | undefined> = { ...process.env };
  for (const k of Object.keys(env)) if (k.startsWith("GIT_CONFIG_")) delete env[k];
  return env;
};

/** `Bun.spawnSync` for git. A spawn that throws is rethrown with the original as `cause` and a one-line message that says where and with which PATH it ran. */
export function spawnGit(cmd: string[], opts: { cwd?: string; env?: Record<string, string | undefined>; stdin?: "ignore" | Blob; stdout: "pipe"; stderr: "pipe" | "ignore" }): { exitCode: number | null; stdout: Buffer; stderr: Buffer } {
  try {
    return Bun.spawnSync(cmd, opts) as { exitCode: number | null; stdout: Buffer; stderr: Buffer };
  } catch (e) {
    const cwd = opts.cwd;
    let here: string;
    try { here = process.cwd(); } catch (err) { here = `unreadable (${(err as Error).message})`; }
    const PATH = (opts.env ?? process.env).PATH;
    const found = Bun.which("git", { PATH: PATH ?? "" });
    const text = (e as Error).message.replace(/\s+/g, " ");
    throw new Error(
      `git could not start (${text}). The cwd passed is ${cwd === undefined ? "none" : JSON.stringify(cwd)}` +
      `${cwd === undefined ? "" : ` and ${existsSync(cwd) ? "exists" : "does not exist"} (existsSync ${existsSync(cwd)})`}. ` +
      `process.cwd() is ${here}. The PATH passed is ${JSON.stringify(PATH)}. Bun.which finds git at ${JSON.stringify(found)}. ` +
      `TMPDIR is ${JSON.stringify(process.env.TMPDIR)} and tmpdir() ${existsSync(tmpdir()) ? "exists" : "does not exist"} (existsSync ${existsSync(tmpdir())}).`,
      { cause: e },
    );
  }
}

export function git(cwd: string, ...args: string[]): { ok: boolean; out: string } {
  return gitWith({}, cwd, ...args);
}

function gitWith(o: { env?: Record<string, string>; stdin?: string | Uint8Array }, cwd: string, ...args: string[]): { ok: boolean; out: string } {
  // Several facts gates write the vault in the same second; a git call that
  // meets another's index.lock waits and tries again (6 facts lost to it, 1 to 3 Nov).
  for (let i = 0; ; i++) {
    const p = spawnGit(["git", ...args], {
      cwd, env: { ...baseEnv(), ...o.env }, stdin: o.stdin === undefined ? "ignore" : new Blob([o.stdin as BlobPart]), stdout: "pipe", stderr: "pipe",
    });
    const r = { ok: p.exitCode === 0, out: (p.stdout.toString() + (p.exitCode === 0 ? "" : p.stderr.toString())).trimEnd() };
    if (r.ok || !r.out.includes("index.lock") || i === 10) return r;
    Bun.sleepSync(200 * (i + 1));
  }
}

/** `git` in `cwd`, not ok (and no throw) when `cwd` does not exist. */
export function probe(cwd: string, ...args: string[]): { ok: boolean; out: string } {
  return existsSync(cwd) ? git(cwd, ...args) : { ok: false, out: "" };
}

/** The ref of a repo's main to read: `origin/<main>` when the repo has it, else the local `<main>`. */
export function mainRef(cwd: string, main: string): string {
  return git(cwd, "rev-parse", "--verify", "-q", `origin/${main}^{commit}`).ok ? `origin/${main}` : main;
}

/** The pause between two tries of a step that met a held git lock: `sleep` waits, Bun.sleepSync unless the caller passes Bun.sleep and awaits it. A test replaces it to free the lock. */
export const lockRetry = { pause: (sleep: (ms: number) => unknown = Bun.sleepSync): unknown => sleep(100 + Math.random() * 200) };

type Run = { ok: boolean; out: string };

/** The tries of a fetch: it yields the number of each try and takes its result. A try after the first waits out the ref lock that another process holds (up to 5 tries). */
function* fetchTries(): Generator<number, Run, Run> {
  let r = yield 0;
  for (let i = 1; i < 5 && !r.ok && r.out.includes("cannot lock ref"); i++) r = yield i;
  return r;
}

/** `git fetch -q origin <refs>`, tried again (up to 5 times) when another process holds the ref lock. Returns the last result. */
export function fetchOrigin(cwd: string, ...refs: string[]): Run {
  const tries = fetchTries();
  for (let t = tries.next(); ; ) {
    if (t.done) return t.value;
    if (t.value) lockRetry.pause();
    t = tries.next(git(cwd, "fetch", "-q", "origin", ...refs));
  }
}

/** fetchOrigin that does not block: each try is an async spawn and each pause an async sleep. */
export async function fetchOriginAsync(cwd: string, ...refs: string[]): Promise<Run> {
  const tries = fetchTries();
  for (let t = tries.next(); ; ) {
    if (t.done) return t.value;
    if (t.value) await lockRetry.pause(Bun.sleep);
    let p;
    // A spawn throws when cwd is gone: the UI's timer then fetches a store a test removed.
    try { p = Bun.spawn(["git", "fetch", "-q", "origin", ...refs], { cwd, env: baseEnv(), stdin: "ignore", stdout: "ignore", stderr: "pipe" }); }
    catch (e) { return { ok: false, out: (e as Error).message }; }
    const [err, code] = await Promise.all([new Response(p.stderr).text(), p.exited]);
    t = tries.next({ ok: code === 0, out: code === 0 ? "" : err.trimEnd() || `git fetch exited ${code}` });
  }
}

/** `tree` with each path in `files` set to its content, or removed when it is null. */
export function treeWithFiles(cwd: string, tree: string, files: Record<string, string | Uint8Array | null>): string {
  const index = join(tmpdir(), `aleph-index-${process.pid}-${crypto.randomUUID()}`);
  const env = { GIT_INDEX_FILE: index };
  try {
    const steps = [["read-tree", tree]];
    for (const [path, content] of Object.entries(files)) {
      if (content === null) { steps.push(["update-index", "--force-remove", path]); continue; }
      const blob = gitWith({ stdin: content }, cwd, "hash-object", "-w", "--stdin");
      if (!blob.ok) throw new Error(`hash-object: ${blob.out}`);
      steps.push(["update-index", "--add", "--cacheinfo", `100644,${blob.out},${path}`]);
    }
    for (const args of steps) {
      const r = gitWith({ env }, cwd, ...args);
      if (!r.ok) throw new Error(`${args[0]}: ${r.out}`);
    }
    const w = gitWith({ env }, cwd, "write-tree");
    if (!w.ok) throw new Error(`write-tree: ${w.out}`);
    return w.out;
  } finally {
    rmSync(index, { force: true });
  }
}

/**
 * Fast-forward the main checkout to origin/<main>. Returns why it could not,
 * or null. It never moves a checkout that is on another branch or has tracked changes.
 */
export function followMain(path: string, main: string): string | null {
  const head = git(path, "symbolic-ref", "--short", "HEAD").out;
  if (head !== main) return `main checkout not updated: it is on ${head || "a detached HEAD"}`;
  const dirty = git(path, "status", "--porcelain", "--untracked-files=no").out.split("\n").filter(Boolean).map((l) => l.slice(3));
  if (dirty.length) return `main checkout not updated: it has changes in ${dirty.join(", ")}`;
  const ff = git(path, "merge", "--ff-only", "-q", `origin/${main}`);
  return ff.ok ? null : `main checkout not updated: ${ff.out.split("\n")[0]}`;
}

/**
 * Hooks inherit the shell, not a dotenv; read the dotenv file (`envFile()`) ourselves. process.env wins.
 * Only LANGFUSE_* keys, plus any `extra` names the caller asks for (the UI server asks for its password): the file also holds model API keys, and a nested `claude -p`
 * that inherits ANTHROPIC_API_KEY bills the API instead of the subscription.
 */
export function loadDotenv(path = envFile()): void {
  let text: string;
  try { text = readFileSync(path, "utf8"); } catch { return; }
  for (const raw of text.split("\n")) {
    const line = raw.trim();
    if (!line || line.startsWith("#")) continue;
    const eq = line.indexOf("=");
    if (eq < 1) continue;
    const key = line.slice(0, eq).trim();
    if (!key.startsWith("LANGFUSE_")) continue;
    let value = line.slice(eq + 1).trim();
    if ((value.startsWith('"') && value.endsWith('"')) || (value.startsWith("'") && value.endsWith("'"))) value = value.slice(1, -1);
    if (process.env[key] === undefined) process.env[key] = value;
  }
}

export function langfuseConfig(): LangfuseConfig | null {
  loadDotenv();
  const publicKey = process.env.LANGFUSE_PUBLIC_KEY;
  const secretKey = process.env.LANGFUSE_SECRET_KEY;
  if (!publicKey || !secretKey) return null;
  return { baseUrl: (process.env.LANGFUSE_BASE_URL ?? "http://127.0.0.1:3010").replace(/\/+$/, ""), publicKey, secretKey };
}

/**
 * Langfuse environment for the session: `ALEPH_ENV` if set (the live tests
 * say `test`), else `headless` when an ancestor is `claude -p`, else `interactive`.
 * The parent chain is the only place a hook can see how claude was started.
 */
export function classifyEnvironment(ancestorArgvs: string[][], override = process.env.ALEPH_ENV): string {
  if (override) return override;
  // claude is argv[0], or argv[1] behind a shell wrapper; only flags after it count
  const headless = ancestorArgvs.some((argv) => {
    const at = argv.slice(0, 2).findIndex((arg) => arg === "claude" || arg.endsWith("/claude"));
    return at >= 0 && argv.slice(at + 1).some((arg) => arg === "-p" || arg === "--print");
  });
  return headless ? "headless" : "interactive";
}

const DEPTH = 6;

function ancestorArgvs(): string[][] {
  const out: string[][] = [];
  let pid = process.ppid;
  for (let i = 0; i < DEPTH && pid > 1; i++) {
    let cmdline: string, status: string;
    try { cmdline = readFileSync(`/proc/${pid}/cmdline`, "utf8"); status = readFileSync(`/proc/${pid}/status`, "utf8"); } catch { break; }
    out.push(cmdline.split("\0").filter(Boolean));
    pid = Number(status.match(/^PPid:\s*(\d+)/m)?.[1] ?? 0);
  }
  return out;
}

export function sessionEnvironment(): string {
  return classifyEnvironment(ancestorArgvs());
}

// Local time, not UTC. `toISOString` stamps the UTC date, so west of Greenwich
// every issue filed in the evening carried tomorrow's date.
export const today = (d = new Date()): string =>
  `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;

/** The local time of `d` as zero-padded `HH:MM`. */
export const clock = (d = new Date()): string =>
  `${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")}`;

/** The local date `n` calendar days after `day` (YYYY-MM-DD). Noon keeps a DST change from moving it, which `n` times 24 hours of ms does not. */
export function addDays(day: string, n: number): string {
  const d = new Date(`${day}T12:00:00`);
  d.setDate(d.getDate() + n);
  return today(d);
}

/** The local date `n` calendar days before `from`. */
export const daysAgo = (n: number, from = new Date()): string => addDays(today(from), -n);

/** The local date before `now`. */
export const yesterday = (now: number): string => addDays(today(new Date(now)), -1);
