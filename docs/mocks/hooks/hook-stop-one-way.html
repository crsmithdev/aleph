/**
 * The refit-code Stop hook: the tool guards of the skill. It asks three questions of the current change, with no model:
 *
 *   co-change   a file that history says changes with a changed file (confidence ≥ 0.7) is left out
 *   boundaries  a changed file holds a copy of code that `.jscpd.json` keeps inside its owner
 *   new copies  a changed file holds a copy of code (renamed copies too) that the branch's base did not have
 *
 * A finding blocks the stop once per session, with the evidence, so the agent changes the companion, calls the
 * owner, or records the second way in `.refit.json`. The same findings do not block twice, so a retry after its
 * own block passes. A session blocks at most 3 times in all; after that the hook stays silent. A stop that another Stop hook already continued is checked like a first stop. A change is the agent's
 * when it came after the session's start and the session's transcript names its file in an edit or a command, so a
 * commit by another session and a file another process wrote are not counted. A boundary copy counts only when its
 * lines are not in the base. A session with no start record is skipped, and so is a factory agent.
 */

/** The co-change rules of the commits at HEAD: history and rules cost about 240 ms, so one file per repo keeps the rules for the HEAD they came from. */
function headRules(root: string): Map<string, Rule[]> {
  const r = probe(root, "rev-parse", "HEAD");
  if (!r.ok) return new Map(); // no commit yet: no history to learn from
  const head = r.out.trim();
  const file = join(spoolDir(), `cochange-${createHash("sha1").update(root).digest("hex").slice(0, 12)}.json`);
  const kept = readJsonOr<{ head: string; rules: [string, Rule[]][] } | null>(file, null);
  if (kept?.head === head) return new Map(kept.rules);
  const all = rules(history(root));
  writeJson(file, { head, rules: [...all] });
  return all;
}

/** The `scan` of the working tree against `from`: jscpd takes seconds, so one file per repo keeps its result for the tree it came from (HEAD, the uncommitted diff and the untracked files) and the base. */
function treeScan(root: string, from: string): ReturnType<typeof scan> {
  const untracked = probe(root, "ls-files", "--others", "--exclude-standard", "-z").out.split("\0").filter(Boolean);
  const tree = [from, probe(root, "rev-parse", "HEAD").out, probe(root, "diff", "HEAD").out, untracked.join("\0"), untracked.length ? probe(root, "hash-object", "--", ...untracked).out : ""];
  const key = createHash("sha1").update(tree.join("\0")).digest("hex");
  const file = join(spoolDir(), `jscpd-${createHash("sha1").update(root).digest("hex").slice(0, 12)}.json`);
  const kept = readJsonOr<{ key: string; found: ReturnType<typeof scan> } | null>(file, null);
  if (kept?.key === key) return kept.found;
  const found = scan(root, from || undefined);
  writeJson(file, { key, found });
  return found;
}

/** The block reason for the findings, or null when there are none. */
function reason(root: string, since: number, mine: Set<string>): string | null {
  const from = base(root);
  const change = new Set([...changed(root, from, since)].filter((f) => mine.has(f)));
  if (!change.size) return null;
  const ok = accepted(root);
  const left = holes(root, change, headRules(root), ok);
  const { clones, violations } = treeScan(root, from);
  const copies = violations.filter((v) => change.has(v.copy.name) && added(root, from, v.copy));
  const pasted = clones.filter((c) => c.isNew && (change.has(c.a.name) || change.has(c.b.name)) && !violations.some((v) => (v.copy.name === c.a.name && v.owner.name === c.b.name) || (v.copy.name === c.b.name && v.owner.name === c.a.name)) && !isAccepted(ok, c.a.name, c.b.name));
  if (!left.length && !copies.length && !pasted.length) return null;
  const lines = [
    ...left.map((h) => `- ${h.a} changed; ${h.b} changed with it in ${h.co} of ${h.n} commits (confidence ${h.conf.toFixed(2)}, e.g. ${h.shas.join(" ")}). Change ${h.b} too, or record the pair in .refit.json.`),
    ...copies.map((v) => `- ${at(v.copy)} copies ${at(v.owner)}, which boundary "${v.rule}" keeps in its owner. Call the owner instead.`),
    ...pasted.map((c) => `- ${c.b.name}:${c.b.start}-${c.b.end} repeats ${c.a.name}:${c.a.start}-${c.a.end} (${c.lines} lines), a copy the base did not have. Call or extend the first, or record the pair in .refit.json.`),
  ];
  return `Refit: this change may add a second way or leave a copy behind.\n${lines.join("\n")}\nA second way on purpose goes in .refit.json as {"copies": [...], "reason": "..."} under "accepted".`;
}

/** The pause between two tries of a step that met a held git lock: `sleep` waits, Bun.sleepSync unless the caller passes Bun.sleep and awaits it. A test replaces it to free the lock. */
export const lockRetry = { pause: (sleep: (ms: number) => unknown = Bun.sleepSync): unknown => sleep(100 + Math.random() * 200) };

type Run = { ok: boolean; out: string };

/** The tries of a fetch: it yields the number of each try and takes its result. A try after the first waits out the ref lock that another process holds (up to 5 tries). */
function* fetchTries(): Generator<number, Run, Run> {
  let r = yield 0;
  for (let i = 1; i < 5 && !r.ok && r.out.includes("cannot lock ref"); i++) r = yield i;
  return r;
}

/** `git fetch -q origin <refs>`, tried again (up to 5 times) when another process holds the ref lock. Returns the last result. */
export function fetchOrigin(cwd: string, ...refs: string[]): Run {
  const tries = fetchTries();
  for (let t = tries.next(); ; ) {
    if (t.done) return t.value;
    if (t.value) lockRetry.pause();
    t = tries.next(git(cwd, "fetch", "-q", "origin", ...refs));
  }
}

/** fetchOrigin that does not block: each try is an async spawn and each pause an async sleep. */
export async function fetchOriginAsync(cwd: string, ...refs: string[]): Promise<Run> {
  const tries = fetchTries();

const JSCPD = jscpd(join(import.meta.dir, "../.."));

// A statement that imports, or re-exports from, a module; it may span lines.
const IMPORT = /^(?:import\b[^;'"]*?|export\b[^;'"]*?\bfrom\s*)["'][^"']*["'];?[ \t]*$/gm;

/** Whether every line of `side` is a blank line or part of an import statement. */
function isImports(root: string, side: { name: string; start: number; end: number }): boolean {
  const text = readFileSync(join(root, side.name), "utf8");
  const lines = text.split("\n");
  const imported = new Set<number>();
  for (const m of text.matchAll(IMPORT)) {
    const first = text.slice(0, m.index).split("\n").length;
    for (let n = first; n < first + m[0].split("\n").length; n++) imported.add(n);
  }
  return lines.slice(side.start - 1, side.end).every((line, i) => !line.trim() || imported.has(side.start + i));
}

type Side = { name: string; start: number; end: number };
export type Clone = { a: Side; b: Side; lines: number; isNew: boolean };
type Violation = { rule: string; copy: Side & { function?: string }; owner: Side & { function?: string } };

/** The clones and boundary violations of the working tree, or nothing when the repo has no `.jscpd.json`. */
export function scan(root: string, base?: string): { clones: Clone[]; violations: Violation[] } {
  if (!existsSync(join(root, ".jscpd.json"))) return { clones: [], violations: [] };
  const out = mkdtempSync(join(tmpdir(), "refit-"));
  try {
    const run = Bun.spawnSync([process.execPath, JSCPD, "--reporters", "json", "--output", out, "--silent", ...(base ? ["--baseline-from-ref", base] : []), "."], { cwd: root, stdout: "ignore", stderr: "pipe" });
    const file = join(out, "jscpd-report.json");
    if (!existsSync(file)) throw new Error(`jscpd exited ${run.exitCode} and wrote no report: ${run.stderr}`);
    const report = JSON.parse(readFileSync(file, "utf8"));
    const side = (f: any) => ({ name: f.name, start: f.start, end: f.end, ...(f.function ? { function: f.function } : {}) });
    const violations = (report.boundaries ?? []).flatMap((b: any) => b.violations.map((v: any) => ({ rule: b.name, copy: side(v.copy), owner: side(v.protected) })));
    // jscpd exits 1 when it finds a boundary violation; any other failure is an error.
    if (run.exitCode !== 0 && !violations.length) throw new Error(`jscpd exited ${run.exitCode}: ${run.stderr}`);
    const clones = report.duplicates.map((d: any) => ({ a: side(d.firstFile), b: side(d.secondFile), lines: d.lines, isNew: d.isNew === true }));
    return { clones: clones.filter((c: Clone) => !(isImports(root, c.a) && isImports(root, c.b))), violations };
  } finally {
    rmSync(out, { recursive: true, force: true });
  }
}

/** Whether the lines of `side` are new since `base`: jscpd marks no violation as new, and its `isNew` on a clone follows the order of equal clones, not the file. A copy that the base file holds verbatim is old, wherever it sits in the file. */
export function added(root: string, base: string, side: { name: string; start: number; end: number }): boolean {
  const was = base ? probe(root, "show", `${base}:${side.name}`) : null;
  if (!was?.ok) return true;
  const copy = readFileSync(join(root, side.name), "utf8").split("\n").slice(side.start - 1, side.end).join("\n").trimEnd();
  return !was.out.includes(copy);
}

#!/usr/bin/env bun
/**
 * Co-change from git history: which files change together, and which files the history says are fixed again and
 * again. The seed adds jscpd's copies of code (clones.ts). The find step seeds its hunt with it (`seed`); the Stop hook asks it which companions a change left out (`holes`).
 *
 * A rule A → B says: a commit that changes A also changes B. Its confidence is the Wilson lower bound of
 * co(A,B) / n(A), so a pair seen 3 times of 3 does not outrank one seen 40 times of 44. Merges and commits of more
 * than 30 files teach nothing about pairs and are skipped, and so are test files.
 *
 *   bun cochange.ts seed  [--since 90.days]   JSON: the files fixed most, and the strongest pairs
 *   bun cochange.ts holes                      JSON: companions the current change left out
 */
import { existsSync, readFileSync, statSync } from "node:fs";
import { join, resolve } from "node:path";
import { digest } from "../../hooks/lib/digest.ts";
import { probe } from "../../lib/git.ts";
import { readLines } from "../../lib/jsonl.ts";
import { expand } from "../../lib/paths.ts";
import { isTestFile } from "../../lib/test-files.ts";
import { scan } from "./clones.ts";

const WIDE = 30;
const SUPPORT = 10;
const FLOOR = 0.6;
const BLOCK = 0.7;
// Design drafts and browser snapshots are not code: jscpd ignores them (.jscpd.json) and so does the change set.
const DRAFTS = /^(docs\/mocks|\.playwright-cli)\//;
const FIX = /\b(fix|fixes|fixed|stale|mismatch|wrong|again|drift)\b/i;

export type Commit = { sha: string; subject: string; files: string[] };
export type Rule = { a: string; b: string; co: number; n: number; conf: number; shas: string[] };
export type Accepted = { concept?: string; copies: string[]; reason: string };

/** The output of `git`, or a throw with git's message: the hook reports it, so a failed git is not a quiet pass. */
function git(cwd: string, ...args: string[]): string {
  const r = probe(cwd, ...args);
  if (!r.ok) throw new Error(`git ${args[0]} failed: ${r.out}`);
  return r.out;
}

/** The merge-base of HEAD and `ref`, or "" when there is none or `ref` is not there. */
const mergeBase = (root: string, ref: string): string => { const r = probe(root, "merge-base", "HEAD", ref); return r.ok ? r.out.trim() : ""; };

/** The last `limit` commits that are not merges, newest first. Each file is named by the path it has today, so a
 * rename does not split a file's history in two: walking newest first, a rename teaches the old path its new name. */
export function history(cwd: string, limit = 2000, since?: string): Commit[] {
  const out = git(cwd, "log", "--no-merges", "-M", "--name-status", "--format=\x1e%h\x1f%s", `-n${limit}`, ...(since ? [`--since=${since}`] : []));
  const today = new Map<string, string>();
  return out.split("\x1e").slice(1).map((c) => {
    const [head, ...lines] = c.split("\n");
    const [sha, subject] = head.split("\x1f");
    const files = lines.filter(Boolean).map((l) => {
      const [status, a, b] = l.split("\t");
      const path = status[0] === "R" ? b : a;
      const name = today.get(path) ?? path;
      if (status[0] === "R") today.set(a, name);
      return name;
    });
    return { sha, subject, files };
  });
}

const Z = 1.96;

/** The Wilson lower bound of k successes in n trials, at 95%. */
export function wilson(k: number, n: number): number {
  if (n === 0) return 0;
  const p = k / n, z2 = Z * Z;
  return (p + z2 / (2 * n) - Z * Math.sqrt((p * (1 - p) + z2 / (4 * n)) / n)) / (1 + z2 / n);
}

/** Every rule A → B with n(A) ≥ 10 and confidence ≥ 0.6, keyed by A. */
export function rules(commits: Commit[]): Map<string, Rule[]> {
  const n = new Map<string, number>();
  const co = new Map<string, { k: number; shas: string[] }>();
  for (const c of commits) {
    if (c.files.length > WIDE) continue;
    // A test changes with its code by design; it is not a second copy of a concept.
    const files = c.files.filter((f) => !isTestFile(f));
    for (const a of files) {
      n.set(a, (n.get(a) ?? 0) + 1);
      for (const b of files) {
        if (a === b) continue;
        const e = co.get(`${a}\0${b}`) ?? { k: 0, shas: [] };
        e.k++;
        if (e.shas.length < 3) e.shas.push(c.sha);
        co.set(`${a}\0${b}`, e);
      }
    }
  }
  const out = new Map<string, Rule[]>();
  for (const [key, e] of co) {
    const [a, b] = key.split("\0");
    const na = n.get(a)!;
    if (na < SUPPORT) continue;
    const conf = wilson(e.k, na);
    if (conf < FLOOR) continue;
    out.set(a, [...(out.get(a) ?? []), { a, b, co: e.k, n: na, conf, shas: e.shas }]);
  }
  return out;
}

/** The commit where the branch left the default branch, or "" when there is none. */

/** The files the current change touches: the branch since it left the default branch, plus the working tree. With `since` (ms), only the commits and edits from then on, so a change from before the session does not count. */
export function changed(root: string, from = base(root), since = 0): Set<string> {
  const branch = !from ? [] : since
    ? git(root, "log", `--since=${new Date(since).toISOString()}`, "--name-only", "--pretty=format:", `${from}..HEAD`).split("\n")
    : git(root, "diff", "--name-only", `${from}...HEAD`).split("\n");
  const edited = (f: string) => { try { return statSync(join(root, f)).mtimeMs >= since; } catch { return true; } };
  const tree = git(root, "status", "--porcelain", "--untracked-files=all").split("\n").map((l) => l.slice(3).split(" -> ").pop()!).filter(edited);
  return new Set([...branch, ...tree].filter((f) => f && !DRAFTS.test(f)));
}

/**
 * The files of `root` that the session's edits and commands name, from its whole transcript. An Edit or Write names its file;
 * a Bash command (`sed -i`, a heredoc, `git checkout -- f`) names it as a word, which resolves from the cwd of that call.
 * A command that writes a file it does not name (a glob, a script, `git add -A`) is missed.
 */
export function touched(transcript: string, cwd: string, root: string): Set<string> {
  const out = new Set<string>();
  const ids = new Set(readLines(transcript).map((e) => e.promptId).filter(Boolean));
  for (const id of ids) {
    for (const [target, dir] of digest(transcript, id).dirs) {
      for (const word of target.split(/[\s'"`<>|;&()=,]+/)) {
        const abs = resolve(dir ?? cwd, expand(word));
        if (abs.startsWith(`${root}/`)) out.add(abs.slice(root.length + 1));
      }
    }
  }
  return out;
}

/** For each file a rule predicts and the change left out, the strongest rule that predicts it. */
export function holes(root: string, change: Set<string>, all: Map<string, Rule[]>, ok: Accepted[], floor = BLOCK): Rule[] {
  const best = new Map<string, Rule>();
  for (const a of change) {
    for (const r of all.get(a) ?? []) {
      if (r.conf < floor || change.has(r.b) || !existsSync(join(root, r.b)) || isAccepted(ok, a, r.b)) continue;
      if ((best.get(r.b)?.conf ?? 0) < r.conf) best.set(r.b, r);
    }
  }
  return [...best.values()].sort((x, y) => y.conf - x.conf);
}

  const fixes = new Map<string, string[]>();
  for (const c of commits) if (FIX.test(c.subject)) for (const f of c.files) fixes.set(f, [...(fixes.get(f) ?? []), `${c.sha} ${c.subject}`]);
  const pairs = [...rules(commits).values()].flat().sort((x, y) => y.conf - x.conf);
  return {
    fixes: [...fixes].sort((x, y) => y[1].length - x[1].length).slice(0, 25).map(([file, cs]) => ({ file, count: cs.length, commits: cs.slice(0, 5) })),
    pairs: pairs.slice(0, 25).map(({ a, b, co, n, conf }) => ({ a, b, co, n, conf: Math.round(conf * 100) / 100 })),

  };
}
