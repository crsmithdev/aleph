#!/usr/bin/env bun
/**
 * Every observability hook is this one script. It reads the event from stdin,
 * turns it into one OTLP span, and posts it to Langfuse. Wired with
 * `async: true` (except Stop, StopFailure and SessionEnd, which the session
 * would not wait for), so it never blocks a turn and its stdout is ignored.
 *
 * One trace per turn, all in one Langfuse session, so the Sessions page shows
 * a session as its prompts and replies:
 *
 *   turn (root)  →  tool | agent → tool | generation | event
 */
import { langfuseConfig, sessionEnvironment } from "../lib/env.ts";
import { scopeOfCwd } from "../lib/paths.ts";
import { agent, prune, session, tool, turn } from "./lib/handshake.ts";
import { attrs, nano, postScore, postSpans, spanId, truncate, turnSpanIdFor, turnTraceIdFor, type AttrValue, type Score, type Span } from "../lib/otlp.ts";
import { parseRating } from "./lib/rating.ts";
import { traceAttrs } from "./lib/trace.ts";
import { readEntries } from "./lib/digest.ts";
import { readLines } from "../lib/jsonl.ts";
import { decide, judgeChoice, type Question } from "../lib/jev.ts";
import { costDetails, type Usage } from "../lib/pricing.ts";

type Payload = Record<string, any>;

const input: Payload = JSON.parse(await Bun.stdin.text());
const cfg = langfuseConfig();
const sessionId: string | undefined = input.session_id;
if (!cfg || !sessionId) process.exit(0);

// `ALEPH_NOW` fixes the clock for tests, as in lib/todo.ts `clock`.
const now = process.env.ALEPH_NOW ? Date.parse(process.env.ALEPH_NOW) : Date.now();
const event: string = input.hook_event_name ?? "unknown";
const sessionSlot = session(sessionId);
const turnSlot = input.prompt_id ? turn(input.prompt_id) : null;
const agentSlot = input.agent_id ? agent(input.agent_id) : null;

// SessionStart posts nothing. It leaves the name, tags and cwd that every turn trace repeats.
if (event === "SessionStart") {
  prune();
  const project = input.cwd ? scopeOfCwd(input.cwd) : undefined;
  sessionSlot.set({
    start: now,
    name: project,
    cwd: input.cwd,
    tags: [
      `source:${input.source ?? "unknown"}`,
      `mode:${input.permission_mode ?? "unknown"}`,
      ...(project ? [`project:${project}`] : []),
    ],
  });
  process.exit(0);
}

// A span belongs to its turn's trace. Inside a subagent the agent's handshake remembers which.
const resolvedTraceId = input.prompt_id ? turnTraceIdFor(input.prompt_id) : agentSlot?.get()?.traceId;
if (!resolvedTraceId) process.exit(0);
const traceId: string = resolvedTraceId;
const turnSpanId = input.prompt_id ? turnSpanIdFor(input.prompt_id) : undefined;

/** Tool spans inside a subagent hang off the agent span; everything else off the turn. */
function parentForChild(): string | undefined {
  if (agentSlot) return agentSlot.get()?.spanId;
  return turnSpanId;
}

function scalars(payload: Payload, skip: string[]): Record<string, AttrValue> {
  const out: Record<string, AttrValue> = {};
  for (const [key, value] of Object.entries(payload)) {
    if (skip.includes(key) || value === null || value === undefined) continue;
    if (typeof value === "string" || typeof value === "number" || typeof value === "boolean") out[`langfuse.observation.metadata.${key}`] = value;
  }
  return out;
}

const known = sessionSlot.get();
const base = traceAttrs(input);

function span(name: string, type: string, start: number, extra: Record<string, AttrValue | undefined | null>, opts: { id?: string; parent?: string; error?: string; end?: number } = {}): Span {
  const s: Span = {
    traceId,
    spanId: opts.id ?? spanId(),
    parentSpanId: opts.parent,
    name,
    kind: 1,
    startTimeUnixNano: nano(start),
    endTimeUnixNano: nano(Math.max(opts.end ?? now, start)),
    attributes: attrs({ ...base, "langfuse.observation.type": type, ...extra }),
  };
  if (opts.error) s.status = { code: 2, message: opts.error.slice(0, 200) };
  return s;
}

/**
 * The turn is the trace's root, posted once at the prompt and again, with the
 * same id, when it ends: Langfuse updates the span in place. Its input and
 * output are the trace's, which is what the Sessions page shows.
 */
function turnSpan(start: number, output: string | undefined, error?: string, extra: Record<string, AttrValue | undefined> = {}): Span {
  const prompt = turnSlot!.get()?.input;
  const inputText = prompt === undefined ? undefined : truncate(prompt);
  const outputText = output === undefined ? undefined : truncate(output);
  return span("turn", "agent", start, {
    "langfuse.observation.input": inputText,
    "langfuse.trace.input": inputText,
    "langfuse.observation.output": outputText,
    "langfuse.trace.output": outputText,
    "langfuse.observation.level": error ? "ERROR" : "DEFAULT",
    "langfuse.observation.metadata.prompt_id": input.prompt_id,
    ...extra,
  }, { id: turnSpanId, error });
}

const TASK_TYPES: Record<string, string> = {
  code_edit: "The turn mainly writes or changes source code",
  debug: "The turn mainly finds or fixes the cause of a bug, failure or error",
  research: "The turn mainly reads, searches or investigates to answer a question",
  writing: "The turn mainly writes prose: docs, messages, specs, summaries",
  planning: "The turn mainly plans, designs or breaks down work without doing it",
  ops: "The turn mainly does git, deploys, config or factory operations",
  chat: "The turn is conversation with no work product",
  other: "None of the other kinds fits",
};
const TASK_TYPE_QUESTION: Question = { type: "choice", instructions: "Which kind of turn is this?", criteria: TASK_TYPES };

/** Jev's top task type for the turn and its probability, or {} when Jev is off, has nothing to read, errors or is late. */
async function taskType(prompt: string | undefined, finalMessage: string): Promise<Record<string, AttrValue>> {
  if (!prompt?.trim() || judgeChoice() === "haiku") return {};
  try {
    const d = await decide({ user_prompt: prompt, final_message: finalMessage }, { task_type: TASK_TYPE_QUESTION }, 1500);
    const a = d.answers.task_type;
    if (a.type !== "choice" || !(a.choice in TASK_TYPES)) return {};
    const p = a.probabilities?.[a.choice] ?? a.confidence;
    return {
      "langfuse.trace.metadata.task_type": a.choice,
      ...(typeof p === "number" ? { "langfuse.trace.metadata.task_type_p": p } : {}),
    };
  } catch {
    return {};
  }
}

/** Whether an assistant entry carries the final message. An empty message has nothing to wait for. */
function landed(entries: Record<string, any>[], finalMessage: string): boolean {
  const needle = finalMessage.trim().slice(0, 80);
  return !needle || entries.some((e) => e.type === "assistant" && Array.isArray(e.message?.content)
    && e.message.content.some((block: any) => block.type === "text" && typeof block.text === "string" && block.text.includes(needle)));
}

/**
 * Stop fires while the last assistant entry may still be on its way to disk
 * (measured: the final request was missing from the transcript at Stop). Wait,
 * briefly, until an assistant entry carries the final message. Only the late
 * copy of the hook waits: the Stop hook is synchronous.
 */
function settledEntries(transcriptPath: string, promptId: string, finalMessage: string) {
  for (let attempt = 0; attempt < 20; attempt++) {
    const entries = readEntries(transcriptPath, promptId);
    if (landed(entries, finalMessage)) return entries;
    Bun.sleepSync(100);
  }
  return readEntries(transcriptPath, promptId);
}

/**
 * One generation per API request in these entries. Assistant entries come one per
 * content block and repeat the request's usage, so dedupe on requestId. The
 * span runs from the previous entry's timestamp to this one's: the model was
 * the only thing happening in between.
 */
function generations(entries: Record<string, any>[], parent: string | undefined): Span[] {
  const out: Span[] = [];
  const seen = new Set<string>();
  let previous = 0;
  for (const entry of entries) {
    const at = Date.parse(entry.timestamp ?? "") || previous || now;
    const usage = entry.message?.usage;
    const requestId: string | undefined = entry.requestId;
    if (entry.type === "assistant" && usage && requestId && !seen.has(requestId)) {
      seen.add(requestId);
      const model: string = entry.message?.model ?? "unknown";
      const usageDetails: Usage = { input: usage.input_tokens ?? 0, output: usage.output_tokens ?? 0 };
      if (usage.cache_read_input_tokens) usageDetails.cache_read_input_tokens = usage.cache_read_input_tokens;
      if (usage.cache_creation_input_tokens) usageDetails.cache_creation_input_tokens = usage.cache_creation_input_tokens;
      const cost = costDetails(model, usageDetails);
      out.push(span(model, "generation", previous || at, {
        "langfuse.observation.model.name": model,
        "langfuse.observation.usage_details": JSON.stringify(usageDetails),
        "langfuse.observation.cost_details": cost ? JSON.stringify(cost) : undefined,
        "langfuse.observation.metadata.request_id": requestId,
      }, { parent, end: at }));
    }
    previous = at;
  }
  return out;
}

/**
 * One event per assistant text block in this turn, in transcript order, at the
 * block's timestamp, so the chat view can show what Claude said between tool
 * calls. Thinking blocks are stored empty and are skipped.
 */
function textBlocks(entries: Record<string, any>[]): Span[] {
  const out: Span[] = [];
  let previous = 0;
  for (const entry of entries) {
    const at = Date.parse(entry.timestamp ?? "") || previous || now;
    previous = at;
    if (entry.type !== "assistant" || !Array.isArray(entry.message?.content)) continue;
    for (const block of entry.message.content) {
      if (block.type !== "text" || typeof block.text !== "string" || !block.text.trim()) continue;
      out.push(span("text", "event", at, {
        "langfuse.observation.output": truncate(block.text),
        "langfuse.observation.metadata.seq": out.length,
      }, { parent: turnSpanId, end: at }));
    }
  }
  return out;
}

const spans: Span[] = [];
const scores: Score[] = [];

switch (event) {
  case "UserPromptSubmit": {
    turnSlot!.set({ start: now, input: input.prompt ?? "" });
    spans.push(turnSpan(now, undefined));
    // "N/10" in a prompt rates the previous reply; the session handshake remembers which trace that was
    const previous = known?.lastTraceId;
    sessionSlot.set({ ...(known ?? { start: now }), lastTraceId: traceId });
    const rating = parseRating(input.prompt ?? "");
    if (rating !== null && previous) scores.push({ traceId: previous, name: "rating", value: rating, comment: input.prompt, environment: sessionEnvironment() });
    break;
  }

  case "PreToolUse":
    if (input.tool_use_id) tool(input.tool_use_id).set({ start: now, input: input.tool_input });
    process.exit(0);

  case "PostToolUse":
  case "PostToolUseFailure": {
    const hs = input.tool_use_id ? tool(input.tool_use_id).take() : null;
    const failed = event === "PostToolUseFailure";
    const toolName: string = input.tool_name ?? "tool";
    const name = toolName === "Skill" && input.tool_input?.skill ? `skill:${input.tool_input.skill}` : toolName;
    const output = failed ? input.tool_error : (input.tool_response ?? input.tool_output);
    spans.push(span(name, "tool", hs?.start ?? now, {
      "langfuse.observation.input": truncate(input.tool_input ?? hs?.input ?? {}),
      "langfuse.observation.output": output === undefined ? undefined : truncate(output),
      "langfuse.observation.level": failed ? "ERROR" : "DEFAULT",
      "langfuse.observation.metadata.tool_use_id": input.tool_use_id,
      "langfuse.observation.metadata.tool": toolName,
      "langfuse.observation.metadata.agent_type": input.agent_type,
    }, { parent: parentForChild(), error: failed ? truncate(input.tool_error, 200) : undefined }));
    break;
  }

  case "SubagentStart":
    agentSlot?.set({ start: now, spanId: spanId(), parentSpanId: turnSpanId, traceId });
    process.exit(0);

  case "SubagentStop": {
    const hs = agentSlot?.take();
    const agentSpan = span(input.agent_type ?? "subagent", "agent", hs?.start ?? now, {
      "langfuse.observation.output": truncate(input.last_assistant_message ?? ""),
      "langfuse.observation.metadata.agent_id": input.agent_id,
      "langfuse.observation.metadata.agent_type": input.agent_type,
    }, { id: hs?.spanId, parent: hs?.parentSpanId ?? turnSpanId });
    spans.push(agentSpan);
    if (input.agent_transcript_path) spans.push(...generations(readLines(input.agent_transcript_path), agentSpan.spanId));
    break;
  }

  case "Stop":
  case "StopFailure": {
    // get, not take: a blocked Stop fires again and the turn keeps its start; prune() clears the file later
    const start = turnSlot!.get()?.start ?? now;
    const failed = event === "StopFailure";
    const message: string = input.last_assistant_message ?? "";
    // a failed turn's message is the API error, which never reaches the transcript, so do not wait for it
    if (!failed && input.transcript_path && !process.env.ALEPH_OBS_LATE && !landed(readEntries(input.transcript_path, input.prompt_id), message)) {
      const late = Bun.spawn(["bun", import.meta.path], { detached: true, stdin: "pipe", stdout: "ignore", stderr: "ignore", env: { ...process.env, ALEPH_OBS_LATE: "1", ALEPH_NOW: new Date(now).toISOString() } });
      late.stdin.write(JSON.stringify(input));
      await late.stdin.end();
      late.unref();
      process.exit(0);
    }
    const tag = failed ? {} : await taskType(turnSlot!.get()?.input, message);
    spans.push(turnSpan(start, message, failed ? (input.error ? `${input.error}: ${message}` : message || "StopFailure") : undefined, tag));
    if (input.transcript_path) {
      const entries = settledEntries(input.transcript_path, input.prompt_id, failed ? "" : message);
      spans.push(...generations(entries, turnSpanId));
      if (!failed) spans.push(...textBlocks(entries));
    }
    break;
  }

  default: {
    // PreCompact, PostCompact, PermissionDenied, SessionEnd, anything new that arrives inside a turn.
    spans.push(span(event, "event", now, {
      ...scalars(input, ["session_id", "cwd", "hook_event_name", "transcript_path", "prompt_id"]),
      "langfuse.observation.level": event === "PermissionDenied" ? "WARNING" : "DEFAULT",
      "langfuse.observation.input": input.tool_input ? truncate(input.tool_input) : undefined,
    }, { parent: parentForChild() }));
  }
}

// The turn-ending hooks run synchronously (an async hook is killed when the session exits), so keep their wait short.
const timeoutMs = event === "Stop" || event === "StopFailure" || event === "SessionEnd" ? 3000 : 8000;
const results = await Promise.all([postSpans(cfg, spans, timeoutMs), ...scores.map((score) => postScore(cfg, score, timeoutMs))]);
for (const result of results) if (!result.ok) console.error(`[aleph obs] ${event}: ${result.status ?? ""} ${result.error ?? ""}`.trim());

/** The dotenv file: ALEPH_ENV_FILE, else ~/.aleph/.env. */
const envFile = (): string => alephPath("ALEPH_ENV_FILE", ".env");

/** One key from the dotenv file, never copied to process.env; null when the file or the key is missing. */
function dotenvKey(name: string): string | null {
  let text: string;
  try { text = readFileSync(envFile(), "utf8"); } catch { return null; }
  const m = text.match(new RegExp(`^\\s*(?:export\\s+)?${name}=(.*)$`, "m"));
  return m ? m[1].trim().replace(/^(["'])(.*)\1$/, "$2") : null;
}

/** OPENROUTER_API_KEY from the environment, else that one key from the dotenv file; null when neither has it. */
export function openRouterKey(): string | null {
  return process.env.OPENROUTER_API_KEY || dotenvKey("OPENROUTER_API_KEY");
}

/** ANTHROPIC_API_KEY from the dotenv file only: a process that has it in its environment bills the API instead of the subscription. */
export const anthropicKey = (): string | null => dotenvKey("ANTHROPIC_API_KEY");

/**
 * Hooks inherit the shell, not a dotenv; read the dotenv file (`envFile()`) ourselves. process.env wins.
 * Only LANGFUSE_* keys, plus any `extra` names the caller asks for (the UI server asks for its password): the file also holds model API keys, and a nested `claude -p`
 * that inherits ANTHROPIC_API_KEY bills the API instead of the subscription.
 */
export function loadDotenv(path = envFile()): void {
  let text: string;
  try { text = readFileSync(path, "utf8"); } catch { return; }
  for (const raw of text.split("\n")) {
    const line = raw.trim();
    if (!line || line.startsWith("#")) continue;
    const eq = line.indexOf("=");
    if (eq < 1) continue;
    const key = line.slice(0, eq).trim();
    if (!key.startsWith("LANGFUSE_")) continue;
    let value = line.slice(eq + 1).trim();
    if ((value.startsWith('"') && value.endsWith('"')) || (value.startsWith("'") && value.endsWith("'"))) value = value.slice(1, -1);
    if (process.env[key] === undefined) process.env[key] = value;
  }
}

export function langfuseConfig(): LangfuseConfig | null {
  loadDotenv();
  const publicKey = process.env.LANGFUSE_PUBLIC_KEY;
  const secretKey = process.env.LANGFUSE_SECRET_KEY;
  if (!publicKey || !secretKey) return null;
  return { baseUrl: (process.env.LANGFUSE_BASE_URL ?? "http://127.0.0.1:3010").replace(/\/+$/, ""), publicKey, secretKey };
}

/**
 * Langfuse environment for the session: `ALEPH_ENV` if set (the live tests
 * say `test`), else `headless` when an ancestor is `claude -p`, else `interactive`.
 * The parent chain is the only place a hook can see how claude was started.
 */
export function classifyEnvironment(ancestorArgvs: string[][], override = process.env.ALEPH_ENV): string {
  if (override) return override;
  // claude is argv[0], or argv[1] behind a shell wrapper; only flags after it count
  const headless = ancestorArgvs.some((argv) => {
    const at = argv.slice(0, 2).findIndex((arg) => arg === "claude" || arg.endsWith("/claude"));
    return at >= 0 && argv.slice(at + 1).some((arg) => arg === "-p" || arg === "--print");
  });
  return headless ? "headless" : "interactive";
}

const DEPTH = 6;

function ancestorArgvs(): string[][] {
  const out: string[][] = [];
  let pid = process.ppid;
  for (let i = 0; i < DEPTH && pid > 1; i++) {
    let cmdline: string, status: string;
    try { cmdline = readFileSync(`/proc/${pid}/cmdline`, "utf8"); status = readFileSync(`/proc/${pid}/status`, "utf8"); } catch { break; }
    out.push(cmdline.split("\0").filter(Boolean));
    pid = Number(status.match(/^PPid:\s*(\d+)/m)?.[1] ?? 0);
  }
  return out;
}

export function sessionEnvironment(): string {
  return classifyEnvironment(ancestorArgvs());
}

/**
 * Chris's live store, from the passwd entry of this uid, so a test that sets HOME cannot move it. Bun 1.4.2's
 * `os.userInfo().homedir` follows HOME (Node's does not), so the file is read. Null when no entry is found.
 */
function passwdHome(): string | null {
  const uid = String(process.getuid?.());
  try {
    const row = readFileSync("/etc/passwd", "utf8").split("\n").map((l) => l.split(":")).find((f) => f[2] === uid);
    return row?.[5] || null;
  } catch { return null; }
}
const owner = passwdHome();
const live = owner === null ? null : join(owner, ".aleph");

/**
 * The variable `name`, expanded, else `rel` under ~/.aleph. Under `bun test` (NODE_ENV=test) a path in the live store
 * throws: a test that reaches it, by a missing preload or an env restored to unset, wrote fixture events to the live log.
 */
export function alephPath(name: string, rel: string): string {
  const p = expand(process.env[name] ?? join(home(), ".aleph", rel));
  if (process.env.NODE_ENV === "test" && live !== null && (p === live || p.startsWith(`${live}/`))) throw new Error(`a test reached the live store: ${name} unset, ${p}`);
  return p;
}

/** The registry key of the repo that `cwd` is in, a worktree of it too; else the folder name of the main checkout. With `git` false: the key whose path holds `cwd`, else "", and no git run. */
export function scopeOfCwd(cwd: string, git = true): string {
  touch("registry");
  const repos = readProjects().filter((p) => p.fields.path).map((p) => ({ key: p.name, path: expand(p.fields.path) }));
  const at = (dir: string) => repos.find((r) => dir === r.path || dir.startsWith(`${r.path.replace(/\/+$/, "")}/`));
  if (!git) return at(cwd)?.key ?? "";
  const common = probe(cwd, "rev-parse", "--path-format=absolute", "--git-common-dir");
  const root = common.ok ? dirname(common.out) : cwd;
  return (at(cwd) ?? at(root))?.key ?? basename(root);
}

/**
 * Hooks are separate processes. A Pre* event leaves a small file that the
 * matching Post* event picks up, so a span can carry its real start time and
 * its parent. Files live in ~/.aleph/spool (ALEPH_SPOOL overrides, for tests).
 * Each kind of record has its own typed slot below; the key prefix is the kind.
 */
import { readJsonOr } from "../../lib/json.ts";
import { mkdirSync, readdirSync, statSync, unlinkSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { alephPath } from "../../lib/paths.ts";
import type { Snapshot } from "./snapshot.ts";

export function spoolDir(): string {
  return alephPath("ALEPH_SPOOL", "spool");
}

function pathFor(key: string): string {
  return join(spoolDir(), `${key.replace(/[^A-Za-z0-9_.:-]/g, "_")}.json`);
}

function slot<T>(key: string) {
  const get = (): T | null => {
    return readJsonOr<T | null>(pathFor(key), null);

const MAX_CHARS = 12_000;

export interface Digest {
  prompt: string;
  edits: string[];
  commands: string[];

const EDIT_TOOLS = new Set(["Edit", "Write", "MultiEdit", "NotebookEdit"]);
const TAIL = 500;

function tail(value: unknown, max = TAIL): string {
  const text = typeof value === "string" ? value : JSON.stringify(value) ?? "";
  return text.length <= max ? text : `…${text.slice(-max)}`;
}

/** A Bash result keeps its first and last TAIL chars: a compound command prints its early output first. */
export function bashResult(value: string): string {
  if (value.length <= 2 * TAIL) return value;
  const cut = value.length - 2 * TAIL;
  return `${value.slice(0, TAIL)}…(${cut} chars cut)…${value.slice(-TAIL)}`;
}

function head(value: unknown, max = 200): string {
  const text = typeof value === "string" ? value : JSON.stringify(value) ?? "";
  return text.length <= max ? text : `${text.slice(0, max)}…`;
}

export function resultText(entry: Entry): string {
  const tur = entry.toolUseResult;
  if (tur && typeof tur === "object" && !Array.isArray(tur) && ("stdout" in tur || "stderr" in tur)) {
    return [tur.stdout, tur.stderr ? `stderr: ${tur.stderr}` : ""].filter(Boolean).join("\n");
  }
  const block = Array.isArray(entry.message?.content) ? entry.message.content.find((c: Entry) => c.type === "tool_result") : null;

}

/**
 * Only user entries carry promptId; assistant entries do not. A prompt's turn
 * is the user entry with that id and everything after it until a user entry
 * with a different id.
 */
export function readEntries(transcriptPath: string, promptId: string): Entry[] {
  const entries = readLines(transcriptPath);
  const turn: Entry[] = [];
  let inside = false;
  for (const e of entries) {
    if (e.isSidechain || (e.type !== "user" && e.type !== "assistant")) continue;
    if (e.promptId !== undefined) inside = e.promptId === promptId;
    if (inside) turn.push(e);
  }
  return turn;
}

/**
 * `stopFinal` is the Stop input's last_assistant_message. It wins over the
 * transcript: at Stop the transcript can still lack the last text, so its last
 * text is a mid-turn preamble ("I'll check ...").
 */
export function digest(transcriptPath: string, promptId: string, stopFinal = ""): Digest {
  const entries = readEntries(transcriptPath, promptId);
  const results = resultEntries(entries);

  let prompt = "";
  const edits: string[] = [];
  const commands: string[] = [];
  const dirs: [string, string | undefined][] = [];
  const items: string[] = [];
  let finalMessage = "";

  for (const e of entries) {
    const content = e.message?.content;
    if (e.type === "user") {
      if (e.isMeta) continue;
      if (typeof content === "string") prompt = content;
      else if (Array.isArray(content)) {
        const text = content.filter((c: Entry) => c.type === "text").map((c: Entry) => c.text).join("\n");
        if (text) prompt = text;
      }
      continue;
    }
    if (!Array.isArray(content)) continue;
    const texts: string[] = [];
    for (const block of content) {
      if (block.type === "text" && block.text) texts.push(block.text);
      if (block.type !== "tool_use") continue;
      const name: string = block.name ?? "tool";
      const inp = block.input ?? {};
      const out = results.get(block.id);
      if (EDIT_TOOLS.has(name)) {
        edits.push(inp.file_path ?? "?");
        dirs.push([inp.file_path ?? "?", e.cwd]);
        items.push(`EDIT ${name} ${inp.file_path ?? "?"}`);
      } else if (name === "Bash") {
        commands.push(String(inp.command ?? ""));
        dirs.push([String(inp.command ?? ""), e.cwd]);
        items.push(`RUN ${head(inp.command, 400)}\n  → ${out ? bashResult(resultText(out)) : "(no result)"}`);
      } else {
        items.push(`CALL ${name} ${head(inp)}${out ? `\n  → ${tail(resultText(out), 200)}` : ""}`);
      }
    }
    if (texts.length) finalMessage = texts.join("\n");
  }
  if (stopFinal) finalMessage = stopFinal;

  let body = items.join("\n");
  if (body.length > MAX_CHARS) body = `…(${items.length} items, earliest trimmed)\n${body.slice(-MAX_CHARS)}`;
  const text = [
    `USER PROMPT:\n${head(prompt, 600)}`,
    `TURN (in order):\n${body || "(no tool calls)"}`,
    `FINAL MESSAGE:\n${finalMessage}`,
  ].join("\n\n");
  return { prompt, edits, commands, dirs, items, finalMessage, text };
}

// The one reader of a JSONL transcript. A line that does not parse is skipped: the last line of a live session can be cut off.
import { readFileSync, statSync } from "node:fs";

export type Entry = Record<string, any>;

/** The entries of JSONL `text`, one per line that parses. */
export function jsonLines(text: string): Entry[] {
  const entries: Entry[] = [];
  for (const line of text.split("\n")) {
    if (!line) continue;
    try { entries.push(JSON.parse(line)); } catch { /* partial last line */ }
  }
  return entries;
}

let parsed: { path: string; size: number; entries: Entry[] } | null = null;

/** Every entry of a JSONL transcript, or none when the file cannot be read. A hook process parses a path once while its size holds: a parse of a 25 MB transcript takes about 110 ms. */
export function readLines(transcriptPath: string): Entry[] {
  let size: number;
  try { size = statSync(transcriptPath).size; } catch { return []; }
  if (parsed?.path === transcriptPath && parsed.size === size) return parsed.entries;
  let raw: string;
  try { raw = readFileSync(transcriptPath, "utf8"); } catch { return []; }
  const entries = jsonLines(raw);
  parsed = { path: transcriptPath, size, entries };
  return entries;
}

/** The text of a JSONL transcript, or "" when the file cannot be read. */
export function readText(transcriptPath: string): string {
  try { return readFileSync(transcriptPath, "utf8"); } catch { return ""; }
}

/** The entries of JSONL `text` from the last back. Only a line that matches `grep` is parsed, so a search for a rare entry does not pay a parse of the whole text. */
export function* entriesBackward(text: string, grep: RegExp): Generator<Entry> {
  for (let end = text.length; end > 0;) {
    const start = text.lastIndexOf("\n", end - 1) + 1;
    const line = text.slice(start, end);
    end = start - 1;
    if (!grep.test(line)) continue;
    try { yield JSON.parse(line); } catch { /* partial last line */ }
  }
}

export type Noul = { type: "noul"; instructions: string; criteria?: { true: string; false: string } };
export type Question = { type: "choice"; instructions: string; criteria: Record<string, string> } | Noul;
type Answer =
  | { type: "choice"; choice: string; probabilities?: Record<string, number>; confidence?: number }
  | { type: "noul"; noul: number };
export interface Decision { answers: Record<string, Answer>; cost: number; model: string; ms: number }

/** One request: the state and the questions in, typed answers out. Throws on any failure. */
export async function decide(state: unknown, questions: Record<string, Question>, timeoutMs = 15_000): Promise<Decision> {
  const key = openRouterKey();
  if (!key) throw new Error("no OPENROUTER_API_KEY");
  const started = performance.now();
  const res = await fetch(URL(), {
    method: "POST",
    headers: { authorization: `Bearer ${key}`, "content-type": "application/json" },
    body: JSON.stringify({ model: JEV_MODEL, state, questions }),
    signal: AbortSignal.timeout(timeoutMs),
  });
  const body: any = await res.json().catch(() => null);
  if (!res.ok) throw new Error(`jev ${res.status}: ${JSON.stringify(body?.error ?? body).slice(0, 200)}`);
  for (const id of Object.keys(questions)) {
    const a = body?.answers?.[id];
    if (a?.type !== questions[id].type) throw new Error(`jev: no ${questions[id].type} answer for ${id}`);
  }
  const cost = body.usage?.cost ?? 0;
  // One row per call in `jev`: OpenRouter bills Jev on aleph's key, and its calls are in no transcript, so the OpenRouter rate (models.ts) subtracts them.
  if (cost > 0) {
    try { conn().query("INSERT INTO jev (time, cost) VALUES (?, ?)").run(new Date().toISOString(), cost); } catch { /* a lost row costs cents */ }
  }
  return { answers: body.answers, cost, model: body.model ?? JEV_MODEL, ms: performance.now() - started };
}

export interface Usage { input: number; output: number; cache_read_input_tokens?: number; cache_creation_input_tokens?: number }

/** Longest known prefix wins, so `claude-haiku-4-5-20251001` prices as haiku 4.5. */
export function priceFor(model: string) {
  const key = Object.keys(PER_MILLION).filter((k) => model.startsWith(k)).sort((a, b) => b.length - a.length)[0];
  return key ? PER_MILLION[key] : null;
}

export function costDetails(model: string, usage: Usage): Record<string, number> | null {
  const price = priceFor(model);
  if (!price) return null;
  const prompt = usage.input + (usage.cache_read_input_tokens ?? 0) + (usage.cache_creation_input_tokens ?? 0);
  const p: Rate = price.long && prompt > price.long.above ? price.long.rate : price;
  const cost: Record<string, number> = {
    input: (usage.input * p.input) / 1e6,
    output: (usage.output * p.output) / 1e6,
  };
  if (usage.cache_read_input_tokens) cost.cache_read_input_tokens = (usage.cache_read_input_tokens * p.cacheRead) / 1e6;
  if (usage.cache_creation_input_tokens) cost.cache_creation_input_tokens = (usage.cache_creation_input_tokens * p.cacheWrite) / 1e6;
  cost.total = Object.values(cost).reduce((sum, value) => sum + value, 0);
  return cost;
}
